135    public function applyPolicy(CachePolicy $policy, float $now): self
136    {
137        $metadata = $this
138            ->withTags($policy->tags)
139            ->withVisibility($policy->visibility);
140
141        if (is_int($policy->ttl)) {
142            $ownExpiration = $now + $policy->ttl;
143            $expiration = $policy->clamp && $metadata->expiresAt !== null
144                ? min($metadata->expiresAt, $ownExpiration)
145                : $ownExpiration;

147            return $metadata->withExpiresAt($expiration);
