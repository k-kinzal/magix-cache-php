16/**
17 * Reads and writes storage, bypassing only classified backend failures.
18 *
19 * The Cache port declares its failures as CacheBackendFailure, so a storage
20 * fault always arrives in the RuntimeException family. Each catch takes only
21 * that family and hands the failure to the classifier the boundary declared,
22 * rethrowing everything the classifier does not accept; without a classifier
23 * every failure propagates unchanged. A backend that reports outside the
24 * declared family violates the port contract and propagates as the bug it is.
25 * Entries written under a different storage format are diagnosed as misses,
26 * never served as hits.
27 *
28 * @internal
29 */
30final readonly class GuardedCache
31{
32    /**
33     * Creates a guarded view over the runtime's storage.
34     */
35    public function __construct(
36        private Cache $cache,
37        private ?CacheObserver $observer = null,
38    ) {
39    }
40
41    /**
42     * Returns physically retained data without choosing a reuse policy.
43     *
44     * @template T
45     * @param Closure(): T $typeWitness
46     * @return CacheEntry<T>|null
47     * @throws RuntimeException when the read fails and no classifier accepts it
48     */
49    public function lookup(string $key, ?BackendErrorClassifier $classifier, Closure $typeWitness, float $now): ?CacheEntry
50    {
51        $entry = $this->read($key, $classifier, $typeWitness);
52
53        return $entry !== null && $entry->retainedUntil > $now ? $entry : null;
54    }
55
56    /**
57     * Reads one entry, converting classified failures and corrupt entries to misses.
58     *
59     * @template T
60     * @param Closure(): T $typeWitness
61     * @return CacheEntry<T>|null
62     * @throws RuntimeException when the read fails and no classifier accepts the failure
63     */
64    public function read(string $key, ?BackendErrorClassifier $classifier, Closure $typeWitness): ?CacheEntry
65    {
66        try {
67            $entry = $this->cache->get($key, $typeWitness);
68        } catch (RuntimeException $error) {
69            if ($classifier?->isBackendFailure($error, CacheAccess::Read) !== true) {
70                throw $error;
71            }
72
73            $this->observer?->observe(CacheEvent::BackendBypassed, $key);
74
75            return null;
76        }
77
78        if ($entry !== null && $entry->formatVersion !== CacheEntry::FORMAT_VERSION) {
79            $this->observer?->observe(CacheEvent::CorruptEntry, $key);
80
81            return null;
82        }
83
84        return $entry;
85    }
86
87    /**
88     * Writes one entry, skipping the write on a classified failure.
89     *
90     * @template T
91     * @param CacheEntry<T> $entry
92     * @throws RuntimeException when the write fails and no classifier accepts the failure
93     */
94    public function write(string $key, CacheEntry $entry, ?BackendErrorClassifier $classifier): void
95    {
96        try {
97            $this->cache->set($key, $entry);
98        } catch (RuntimeException $error) {
99            if ($classifier?->isBackendFailure($error, CacheAccess::Write) !== true) {
100                throw $error;
101            }
102
103            $this->observer?->observe(CacheEvent::BackendBypassed, $key);
104
105            return;
106        }
107
108        $this->observer?->observe(CacheEvent::Stored, $key);
109    }
110}
