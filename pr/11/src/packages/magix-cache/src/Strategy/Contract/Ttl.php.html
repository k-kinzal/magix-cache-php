10/**
11 * Declares the lifetime constraint a strategy adds on the normal origin path.
12 *
13 * The contract is about the candidate constraint the operation contributes,
14 * not about the format of an algorithm: a declaration with bounds promises
15 * that the strategy meets one finite lifetime constraint into the produced
16 * metadata, that the constraint lies within the declared bounds relative to
17 * the base time, and that the strategy never extends an expiration a
18 * dependency already imposed. A missing bound is undetermined, not
19 * unlimited. Omitting the attribute, or supplying no arguments, declares
20 * that the operation adds no lifetime constraint at all.
21 * With positional alternatives, the constraint belongs to the union of the
22 * declared points and ranges. Alternatives describe possible outcomes, not simultaneous
23 * constraints, and do not prove which runtime condition selects each one.
24 *
25 * The declaration covers the normal origin path only; a stale answer keeps
26 * the expired expiration it was stored with.
27 */
28#[Attribute(Attribute::TARGET_METHOD)]
29final readonly class Ttl
30{
31    /**
32     * Lower bound of the named range; null for alternatives or no constraint.
33     */
34    public int|ConstructorArg|Arg|null $min;
35
36    /**
37     * Upper bound of the named range; null for alternatives or no constraint.
38     */
39    public int|ConstructorArg|Arg|null $max;
40
41    /**
42     * Whether no lifetime arguments were supplied and no constraint is added.
43     */
44    public bool $unconstrained;
45
46    /**
47     * Normalized positional alternatives; null for named bounds or no constraint.
48     *
49     * @var non-empty-list<int|ConstructorArg|Arg|TtlRange>|null
50     */
51    public ?array $oneOf;
52
53    /**
54     * Declares the candidate lifetime contract of one operation.
55     *
56     * Ttl(30) declares a fixed constraint; Ttl(30, 60, new TtlRange(600, 900))
57     * declares alternatives. Ttl(min: 600, max: 900) is a single range.
58     * Named bounds cannot be mixed with positional alternatives. With no
59     * arguments the operation adds no TTL constraint, as when omitted.
60     *
61     * @param int|ConstructorArg|Arg|TtlRange|null ...$lifetimes Positional fixed lifetimes, references or ranges; alternatively named min/max bounds in seconds. Null is allowed only for an omitted named bound.
62     * @throws InvalidArgumentException when bounds or alternatives are invalid, an unknown name is supplied, or named bounds and positional alternatives are mixed
63     */
64    public function __construct(
65        int|ConstructorArg|Arg|TtlRange|null ...$lifetimes,
66    ) {
67        if (array_is_list($lifetimes)) {
68            $this->min = null;
69            $this->max = null;
70            $this->unconstrained = $lifetimes === [];
71            $this->oneOf = $lifetimes === [] ? null : TtlRange::alternatives($lifetimes);
72
73            return;
74        }
75
76        foreach (array_keys($lifetimes) as $name) {
77            if ($name !== 'min' && $name !== 'max') {
78                throw new InvalidArgumentException('Use positional lifetime alternatives or named min/max bounds, never both.');
79            }
80        }
81
82        $min = $lifetimes['min'] ?? null;
83        $max = $lifetimes['max'] ?? null;
84
85        if ($min instanceof TtlRange || $max instanceof TtlRange) {
86            throw new InvalidArgumentException('A named lifetime bound must be integer seconds or an argument reference.');
87        }
88
89        $range = new TtlRange($min, $max);
90        $this->min = $range->min;
91        $this->max = $range->max;
92        $this->unconstrained = false;
93        $this->oneOf = null;
94    }
95}
