20/**
21 * Applies the composition rules of MagixCache to a statically read boundary.
22 *
23 * The calculator works on TtlEstimate values, so "no constraint" and "not
24 * statically decidable" stay separate all the way to the output. A declared
25 * strategy contributes its candidate constraint the way the runtime meets
26 * it: before the policy is applied, so a derived policy can take its
27 * lifetime from what the strategy contracts guarantee.
28 */
29final readonly class EffectCalculator
30{
31    /**
32     * Returns the constraints the dependencies of a boundary impose on it.
33     *
34     * flatten, zip, sequence and traverse use this same meet of all inputs.
35     * unzip preserves the whole pair's constraints on either projection, so
36     * selecting one side must not remove a discovered dependency here.
37     * An empty collection contributes the unconstrained identity.
38     *
39     * @param list<CacheNode> $children
40     */
41    public function constrain(array $children): DependencyConstraint
42    {
43        $ttl = TtlEstimate::unconstrained();
44        $ttlSource = null;
45        $visibility = Visibility::Shared;
46        $visibilitySource = null;
47        $tags = [];
48        $visibilityUnknown = false;
49        $tagsUnknown = false;
50
51        foreach ($children as $child) {
52            $effect = $child->effect;
53            $visibilityUnknown = $visibilityUnknown || $effect->visibilityUnknown;
54            $tagsUnknown = $tagsUnknown || $effect->tagsUnknown;
55            $tags = array_merge($tags, $effect->tags);
56            $met = $ttl->meet($effect->ttl);
57
58            if (!$met->equals($ttl)) {
59                $ttlSource = $child->boundary->shortId();
60            }
61
62            $ttl = $met;
63
64            if ($effect->visibility->meet($visibility) !== $visibility) {
65                $visibility = $effect->visibility->meet($visibility);
66                $visibilitySource = $child->boundary->shortId();
67            }
68        }
69
70        return new DependencyConstraint($ttl, $ttlSource, $visibility, $visibilitySource, $this->tags($tags), $visibilityUnknown, $tagsUnknown, $children !== []);
71    }

81    /**
82     * Returns the metadata a cached() boundary produces once its policy is applied.
83     */
84    public function applyPolicy(BoundaryDeclaration $boundary, DependencyConstraint $constraint, ?StrategyEffect $strategy = null): CacheEffect
85    {
86        $visibility = $constraint->visibility;
87        $reason = $constraint->visibilitySource === null ? null : 'restricted by '.$constraint->visibilitySource;
88        $scope = $boundary->scope();
89
90        if ($scope->meet($visibility) !== $visibility) {
91            $visibility = $scope->meet($visibility);
92            $reason = 'restricted by a scoped parameter';
93        }
94
95        $policy = $boundary->policy;
96
97        if ($policy === null) {
98            $problem = 'no #[Cache] attribute on the method or its concrete class, so cached() throws a LogicException';
99
100            return new CacheEffect(
101                ttl: TtlEstimate::invalid($problem),
102                visibility: $visibility,
103                tags: $constraint->tags,
104                visibilityReason: $reason,
105                problems: $this->problems([$problem], $strategy),
106                strategy: $strategy,
107            );
108        }
109
110        if ($policy->visibility->meet($visibility) !== $visibility) {
111            $visibility = $policy->visibility->meet($visibility);
112            $reason = 'declared by the policy';
113        }
114
115        $estimate = $this->lifetime($boundary, $policy, $constraint, $strategy);
116        $problems = $this->problems(
117            $estimate->state === TtlEstimateState::Invalid && $estimate->reason !== null ? [$estimate->reason] : [],
118            $strategy,
119        );
120
121        $effect = new CacheEffect(
122            ttl: $estimate,
123            visibility: $visibility,
124            storable: $this->storable($estimate, $visibility, $problems),
125            tags: $this->tags(array_merge($constraint->tags, $policy->tags)),
126            visibilityReason: $reason,
127            problems: $problems,
128            strategy: $strategy,
129        );
130
131        return (new ParameterEffects())->apply($boundary, $constraint, $effect);
132    }

150    /**
151     * Returns the lifetime estimate a policy resolves to for one boundary.
152     *
153     * The candidate constraint of a declared strategy is met into the
154     * upstream constraint first, mirroring the runtime stage order.
155     */
156    public function lifetime(BoundaryDeclaration $boundary, PolicyDeclaration $policy, DependencyConstraint $constraint, ?StrategyEffect $strategy = null): TtlEstimate
157    {
158        $declared = $policy->ttl;
159        $upstream = $constraint->ttl;
160        $candidate = $strategy?->ttl;
161        $combined = $candidate === null ? $upstream : $upstream->meet($candidate);
162        $parameterTtl = (new ParameterEffects())->ttl($boundary);
163        $combined = $parameterTtl === null ? $combined : $combined->meet($parameterTtl);
164        $source = $candidate !== null && !$combined->equals($upstream)
165            ? 'the declared strategy'
166            : ($constraint->ttlSource ?? 'a dependency');
167
168        if ($combined->state === TtlEstimateState::Invalid) {
169            $estimate = $combined;
170        } elseif ($declared === null) {
171            $estimate = TtlEstimate::unknown(condition: 'the declared ttl cannot be read statically');
172        } elseif (is_int($declared)) {
173            $estimate = $this->fixed($declared, $combined, $source);
174        } else {
175            $estimate = $this->derived($declared, $policy->maxTtl, $boundary, $combined, $source, $strategy?->addsConstraint === true || $parameterTtl !== null);
176        }
177
178        if ($candidate !== null && $upstream->state === TtlEstimateState::Unknown && !$upstream->hasFiniteExpiration()
179            && $estimate->state === TtlEstimateState::Unknown && $estimate->reason === null) {
180            $estimate = $estimate->withCondition('an upstream expiration may shorten the lifetime');
181        }
182
183        if ($boundary->hasDynamicTtl) {
184            $estimate = $estimate->meet(TtlEstimate::unknown(
185                condition: 'a #[DynamicTtl] resolver decides the final lifetime at runtime',
186            ));
187        }
188
189        return $parameterTtl === null ? $estimate : $parameterTtl->meet($estimate);
190    }

192    /**
193     * Returns the estimate of a fixed lifetime bounded by its upstream.
194     *
195     * A fixed lifetime is always capped by the upstream expiration and never
196     * fails: with no upstream constraint the declared value stands, and with
197     * an unknown upstream the declared value remains as an upper bound while
198     * a proven lower bound survives.
199     */
200    public function fixed(int $declared, TtlEstimate $upstream, string $source): TtlEstimate
201    {
202        if ($upstream->seconds !== null) {
203            if ($upstream->seconds < $declared) {
204                return TtlEstimate::known($upstream->seconds, 'declared '.$declared.'s, capped by '.$source);
205            }
206
207            return TtlEstimate::known($declared);
208        }
209
210        if ($upstream->state === TtlEstimateState::Unconstrained) {
211            return TtlEstimate::known($declared);
212        }
213
214        $met = TtlEstimate::known($declared)->meet($upstream);
215
216        if ($met->seconds !== null) {
217            return $met->seconds === $declared
218                ? TtlEstimate::known($declared)
219                : TtlEstimate::known($met->seconds, 'declared '.$declared.'s, capped by '.$source);
220        }
221
222        if ($upstream->hasFiniteExpiration() && $met->reason === null) {
223            return $met;
224        }
225
226        return $met->withCondition('an upstream expiration may shorten the declared '.$declared.'s');
227    }

229    /**
230     * Returns the estimate of a lifetime derived from the upstream expiration.
231     *
232     * Ttl::Auto and Ttl::FromUpstream require a finite upstream expiration.
233     * A strategy whose contracts definitely add a finite constraint fulfills
234     * the requirement; otherwise a confirmed missing expiration is an error
235     * unless the boundary itself supplies metadata or resolves a lifetime at
236     * runtime, and an unknown one keeps the requirement as a runtime
237     * condition.
238     */
239    public function derived(Ttl $declared, ?int $maxTtl, BoundaryDeclaration $boundary, TtlEstimate $upstream, string $source, bool $strategyAddsConstraint = false): TtlEstimate
240    {
241        if ($declared === Ttl::FromUpstream && $maxTtl === null) {
242            return TtlEstimate::invalid('Ttl::FromUpstream requires maxTtl, so the declaration cannot be constructed');
243        }
244
245        $cap = $declared === Ttl::FromUpstream ? $maxTtl : null;
246
247        if ($upstream->seconds !== null) {
248            return $cap === null
249                ? TtlEstimate::known($upstream->seconds, 'inherited from '.$source)
250                : TtlEstimate::known(min($upstream->seconds, $cap), 'upstream expiration capped at '.$cap.'s');
251        }
252
253        if ($upstream->state === TtlEstimateState::Unknown) {
254            $capped = $cap === null ? $upstream : $upstream->meet(TtlEstimate::known($cap));
255
256            if ($strategyAddsConstraint || $upstream->hasFiniteExpiration()) {
257                return $capped;
258            }
259
260            return $capped->withCondition('requires a finite upstream expiration at runtime');
261        }
262
263        if ($boundary->suppliesMetadata || $boundary->hasDynamicTtl) {
264            return TtlEstimate::unknown(
265                upperBound: $cap,
266                condition: 'requires the boundary to supply a finite expiration at runtime',
267            );
268        }
269
270        return TtlEstimate::invalid(
271            'Ttl::'.$declared->name.' has no dependency with a finite expiration, so applying the policy throws a LogicException',
272        );
273    }

275    /**
276     * Reports whether the boundary provably stores its result.
277     *
278     * A range with a positive proven lower bound and no runtime condition
279     * stores; a lifetime under a runtime condition cannot be proven to.
280     *
281     * @param list<string> $problems
282     */
283    public function storable(TtlEstimate $estimate, Visibility $visibility, array $problems): bool
284    {
285        $floor = $estimate->seconds
286            ?? ($estimate->state === TtlEstimateState::Unknown && $estimate->reason === null ? $estimate->lowerBound : null);
287
288        return $floor !== null && $floor > 0 && $visibility !== Visibility::NoStore && $problems === [];
289    }
