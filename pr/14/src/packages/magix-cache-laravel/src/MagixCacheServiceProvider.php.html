22     * Registers the runtime against Laravel's PSR-16 repository.
23     */
24    #[Override]
25    public function register(): void
26    {
27        $this->app->singleton(CacheRuntime::class, static function (Container $app): CacheRuntime {
28            $repository = $app->make('cache.store');
29
30            if (!$repository instanceof Repository) {
31                throw new LogicException('Laravel cache.store must implement the cache Repository contract.');
32            }
33
34            return new CacheRuntime(new SimpleCache($repository));
35        });
36    }

38    /**
39     * Fixes the default runtime reference to the container-managed runtime.
40     *
41     * The provider closure resolves through the container on every lookup, so
42     * the registry never holds a runtime instance across requests.
43     */
44    public function boot(): void
45    {
46        if (CacheRuntimeRegistry::isRegistered(CacheRuntimeRegistry::DEFAULT_NAME)) {
47            return;
48        }
49
50        $app = $this->app;
51
52        CacheRuntimeRegistry::register(
53            CacheRuntimeRegistry::DEFAULT_NAME,

55        );
56    }
