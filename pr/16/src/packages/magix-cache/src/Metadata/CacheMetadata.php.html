14/**
15 * Immutable cache constraints that compose only through the fixed meet law.
16 *
17 * The meet is the sole way to add constraints to existing metadata: it selects
18 * the earliest finite expiration, combines cacheability with AND, chooses the
19 * stricter visibility, and unions tags and diagnostic reasons. top() is its
20 * identity, and every composition is as strict as or stricter than each input.
21 */

54     * Creates an immutable cache constraint set.
55     *
56     * @param list<string> $tags
57     * @param list<string> $reasons
58     * @throws InvalidArgumentException when the expiration is not a finite Unix timestamp
59     */

61        ?float $expiresAt = null,
62        bool $cacheable = true,
63        array $tags = [],
64        Visibility $visibility = Visibility::Shared,
65        array $reasons = [],
66    ) {

68            throw new InvalidArgumentException('Expiration must be a finite Unix timestamp or null.');
69        }
70
71        $tokens = new CacheTokenSet();
72        $this->expiresAt = $expiresAt;
73        $this->cacheable = $cacheable;
74        $this->tags = $tokens->tags($tags);
75        $this->visibility = $visibility;
76        $this->reasons = $tokens->reasons($reasons);
77    }

112    /**
113     * Combines this value with every supplied constraint set.
114     */
115    public function meet(self ...$others): self
116    {
117        $result = $this;
118
119        foreach ($others as $other) {
120            $expiresAt = $result->expiresAt === null || $other->expiresAt === null
121                ? $result->expiresAt ?? $other->expiresAt
122                : min($result->expiresAt, $other->expiresAt);
123
124            $result = new self(
125                expiresAt: $expiresAt,
126                cacheable: $result->cacheable && $other->cacheable,
127                tags: array_merge($result->tags, $other->tags),
128                visibility: $result->visibility->meet($other->visibility),
129                reasons: array_merge($result->reasons, $other->reasons),
130            );
131        }
132
133        return $result;
134    }

143            && $this->tags === $other->tags

146    }

148    /**
149     * Reports whether the value is eligible for server-side storage right now.
150     */
151    public function isStorable(float $now): bool
152    {
153        return $this->cacheable
154            && $this->visibility !== Visibility::NoStore
155            && $this->expiresAt !== null
156            && $this->expiresAt > $now;
157    }
