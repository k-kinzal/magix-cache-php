18/**
19 * Derives the composed contract of a declared strategy without running it.
20 *
21 * The resolver binds the #[UseStrategy] arguments to the same create() the
22 * runtime calls, follows the composed constructions in order, binds each
23 * child contract's explicit references to the constructor values it was
24 * built with, and meets the child contracts into one candidate constraint.
25 * A composition never re-declares child contracts; only an explicit
26 * assumption replaces the one analysis item it names. What stays unknown
27 * stays unknown — it is never converted into a default or into "no
28 * constraint".
29 */
30final readonly class StrategyResolver
31{

42    /**
43     * Returns the analyzed strategy effect of a boundary, when one is declared.
44     */
45    public function resolve(BoundaryDeclaration $boundary): ?StrategyEffect
46    {
47        $use = $boundary->useStrategy;
48
49        if ($use === null) {
50            return null;
51        }
52
53        $declaration = $this->declaration($use->strategy);
54
55        if ($declaration === null) {
56            return new StrategyEffect(
57                label: $use->label(),
58                ttl: TtlEstimate::unknown(condition: $use->strategy.' was not found in the scanned sources'),
59            );
60        }
61
62        if (!$declaration->hasCreate) {
63            $problem = $use->strategy.' declares no public static create(), so resolving the strategy throws a LogicException';
64
65            return new StrategyEffect($use->label(), TtlEstimate::invalid($problem), [], null, [$problem]);
66        }
67
68        if ($declaration->definitionProblem !== null) {
69            $problem = $declaration->definitionProblem;
70
71            return new StrategyEffect($use->label(), TtlEstimate::invalid($problem), [], null, [$problem]);
72        }
73
74        [$use, $bindingProblems] = (new ParameterStrategyBinding())->bind($boundary, $use, $declaration);
75        [$environment, $problems] = $this->binding->bindCreate($declaration, $use->arguments);
76        $problems = [...$problems, ...$bindingProblems];
77        $composed = $this->composition($declaration, $environment);
78        $ttl = $problems === [] ? $composed->ttl : TtlEstimate::invalid($problems[0]);
79
80        return new StrategyEffect($use->label(), $ttl, $composed->steps, $composed->addsConstraint, [...$problems, ...$composed->problems]);
80        return new StrategyEffect($use->label(), $ttl, $composed->steps, $composed->addsConstraint, [...$problems, ...$composed->problems], $composed->expirations);
81    }

91    /**
92     * Returns the composed candidate constraint of one bound composition.
93     *
94     * @param array<string, mixed> $environment
95     */
96    public function composition(StrategyDeclaration $declaration, array $environment): StrategyEffect
97    {
98        if ($declaration->definitionProblem !== null) {
99            $problem = $declaration->definitionProblem;
100
101            return new StrategyEffect('', TtlEstimate::invalid($problem), problems: [$problem]);
102        }
103
104        if ($declaration->composed === null) {
105            $condition = $declaration->notes[0] ?? ('the composition of '.$declaration->shortName().' cannot be read statically');
106
107            return new StrategyEffect('', TtlEstimate::unknown(condition: $condition));
108        }
109
110        $ttl = TtlEstimate::unconstrained();
111        $steps = [];
112        $expirations = [];
113        $problems = [];
114        $adds = false;
115        $open = false;
116
117        foreach ($declaration->composed as $instantiation) {
118            [$step, $stepProblems, $stepAdds] = $this->step($declaration, $instantiation, $environment);
119            $steps[] = $step;
120            $expirations = [...$expirations, ...$step->expirations];
121            $problems = [...$problems, ...$stepProblems];
122            $ttl = $ttl->meet($step->ttl);
123            $adds = $adds || $stepAdds === true;
124            $open = $open || $stepAdds === null;
125        }
126
125        return new StrategyEffect('', $ttl, $steps, $adds ? true : ($open ? null : false), $problems);
127        return new StrategyEffect('', $ttl, $steps, $adds ? true : ($open ? null : false), $problems, $expirations);
128    }

130    /**
131     * Returns the analyzed contribution of one composed construction.
132     *
133     * @param array<string, mixed> $environment
134     * @return array{StrategyStep, list<string>, bool|null}
135     */
136    public function step(StrategyDeclaration $declaration, StrategyInstantiation $instantiation, array $environment): array
137    {
138        if ($instantiation->problem !== null) {
139            $problem = $instantiation->problem;
140
141            return [new StrategyStep($instantiation->class, TtlEstimate::invalid($problem)), [$problem], null];
142        }
143
144        $child = $this->declaration($instantiation->class);
145        $assumption = $declaration->assumptionFor($instantiation->class);
146
147        if ($instantiation->viaCreate) {
148            if ($child !== null && $child->hasCreate && $child->composed !== null) {
149                return $this->created($child, $instantiation, $environment);
150            }
151
152            if ($assumption !== null) {
153                return $this->assumed($instantiation->class, $assumption, $environment);
154            }
155
156            $condition = $this->shortName($instantiation->class).'::create() cannot be analyzed statically';
157
158            return [new StrategyStep($instantiation->class, TtlEstimate::unknown(condition: $condition)), [], null];
159        }
160
161        if ($child !== null && !$child->constructible) {
162            $problem = $child->name.' is not a constructible CacheStrategy for StrategyDefinition::of()';
163
164            return [new StrategyStep($child->name, TtlEstimate::invalid($problem)), [$problem], null];
165        }
166
165        if ($child === null || $child->ttl === null) {
167        if ($child === null || ($child->ttl === null && $child->expiration === null)) {
168            if ($assumption !== null) {
169                return $this->assumed($instantiation->class, $assumption, $environment);
170            }
171
172            if ($child === null) {
173                $condition = $instantiation->class.' was not found in the scanned sources';
174
175                return [new StrategyStep($instantiation->class, TtlEstimate::unknown(condition: $condition)), [], null];
176            }
177        }
178
179        return $this->contracted($child, $instantiation, $environment);
180    }

182    /**
183     * Follows a nested composition built by a child create().
184     *
185     * @param array<string, mixed> $environment
186     * @return array{StrategyStep, list<string>, bool|null}
187     */
188    public function created(StrategyDeclaration $child, StrategyInstantiation $instantiation, array $environment): array
189    {
190        $arguments = $this->binding->values($instantiation->arguments, $environment);
191        [$childEnvironment, $problems] = $this->binding->bindCreate($child, $arguments);
192        $composed = $this->composition($child, $childEnvironment);
193        $ttl = $problems === [] ? $composed->ttl : TtlEstimate::invalid($problems[0]);
194
193        return [new StrategyStep($instantiation->class, $ttl), [...$problems, ...$composed->problems], $composed->addsConstraint];
195        return [new StrategyStep($instantiation->class, $ttl, expirations: $composed->expirations), [...$problems, ...$composed->problems], $composed->addsConstraint];
196    }

198    /**
199     * Binds the declared contract of one constructed strategy.
200     *
201     * @param array<string, mixed> $environment
202     * @return array{StrategyStep, list<string>, bool|null}
203     */
204    public function contracted(StrategyDeclaration $child, StrategyInstantiation $instantiation, array $environment): array
205    {
206        $contract = $child->ttl ?? new TtlContract(unconstrained: true);
207
208        [$constructor, $problems] = $this->binding->bindConstructor($child, $instantiation->arguments, $environment);
209        $subject = '#[Ttl] on '.$child->shortName().'::fetch()';
210        [$estimate, $contractProblems] = $this->binding->contract($contract, ContractSource::Constructor, $constructor, $subject);
211        $problems = [...$problems, ...$contractProblems];
212        $expirations = [];
213
214        if ($child->expiration !== null) {
215            [$expiration, $expirationProblems] = (new ExpirationBinding())->resolve($child->expiration, $constructor, '#[ExpiresAt] on '.$child->shortName().'::fetch()');
216            $problems = [...$problems, ...$expirationProblems];
217            $expirations[] = $expiration;
218            $estimate = $estimate->meet(TtlEstimate::unknown(condition: 'duration depends on the origin time and daily expiration', finite: true));
219        }
220
221        if ($problems !== []) {
222            return [new StrategyStep($child->name, TtlEstimate::invalid($problems[0])), $problems, true];
223        }
224
215        return [new StrategyStep($child->name, $estimate), [], !$contract->unconstrained];
225        return [new StrategyStep($child->name, $estimate, expirations: $expirations), [], !$contract->unconstrained || $expirations !== []];
226    }

228    /**
229     * Applies the explicit assumption declared for one composed strategy.
230     *
231     * @param array<string, mixed> $environment
232     * @return array{StrategyStep, list<string>, bool|null}
233     */
234    public function assumed(string $class, TtlAssumption $assumption, array $environment): array
235    {
236        $subject = '#[AssumeTtl] for '.$this->shortName($class);
237        [$estimate, $problems] = $this->binding->contract($assumption->contract(), ContractSource::Create, $environment, $subject);
238
239        if ($problems !== []) {
240            return [new StrategyStep($class, TtlEstimate::invalid($problems[0]), assumed: true), $problems, true];
241        }
242
243        return [new StrategyStep($class, $estimate, assumed: true), [], !$assumption->unconstrained];
244    }
