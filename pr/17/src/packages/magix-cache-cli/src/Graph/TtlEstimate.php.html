14/**
15 * Holds what the analyzer can prove about the lifetime of one boundary.
16 *
17 * The value keeps "no constraint is declared" apart from "the constraint
18 * depends on runtime values": an unknown lifetime is never converted into a
19 * concrete number or into the absence of a constraint. An unknown lifetime
20 * may still carry proven bounds — a lower bound the constraint never goes
21 * under and an upper bound it never exceeds; a missing bound is
22 * undetermined, not unlimited. The analyzer works with durations and
23 * bounds, a different value domain from the runtime's absolute expirations.
24 */
25final readonly class TtlEstimate implements JsonSerializable
26{
27    /**
28     * Creates an estimate; the shape of each state is fixed.
29     *
30     * @param int|null $seconds Determined lifetime, carried only by Known.
31     * @param int|null $upperBound Bound a stored entry cannot outlive, carried only by Unknown.
32     * @param string|null $reason Why the value holds: a derivation for Known, a runtime condition for Unknown, the problem for Invalid.
33     * @param int|null $lowerBound Bound the lifetime never goes under, carried only by Unknown.
34     * @param TtlRangeSet|null $alternatives Disjoint alternatives whose enclosing bounds match this estimate.
35     * @param bool $finite Whether a finite expiration is guaranteed even though its lifetime depends on runtime values.
36     * @throws InvalidArgumentException when a field is combined with a state that cannot carry it
37     */
38    public function __construct(
39        public TtlEstimateState $state,
40        public ?int $seconds = null,
41        public ?int $upperBound = null,
42        public ?string $reason = null,
43        public ?int $lowerBound = null,
44        public ?TtlRangeSet $alternatives = null,
45        public bool $finite = false,
46    ) {
47        if (($state === TtlEstimateState::Known) !== ($seconds !== null)) {
48            throw new InvalidArgumentException('A lifetime in seconds is carried by a Known estimate and nothing else.');
49        }
50
51        if (($upperBound !== null || $lowerBound !== null) && $state !== TtlEstimateState::Unknown) {
52            throw new InvalidArgumentException('A bound is carried by an Unknown estimate and nothing else.');
53        }
54
55        if ($state === TtlEstimateState::Invalid && $reason === null) {
56            throw new InvalidArgumentException('An Invalid estimate must name its problem.');
57        }
58
59        if (($seconds !== null && $seconds < 0) || ($upperBound !== null && $upperBound < 0) || ($lowerBound !== null && $lowerBound < 0)) {
60            throw new InvalidArgumentException('A lifetime must be zero or greater.');
61        }
62
63        if ($lowerBound !== null && $upperBound !== null && $upperBound < $lowerBound) {
64            throw new InvalidArgumentException('A lifetime upper bound cannot be below the lower bound.');
65        }
66
67        if ($alternatives !== null && ($state !== TtlEstimateState::Unknown
68            || $lowerBound !== $alternatives->lowerBound() || $upperBound !== $alternatives->upperBound())) {
69            throw new InvalidArgumentException('Lifetime alternatives require an unknown estimate with matching enclosing bounds.');
70        }
71    }

120    /**
121     * Collapses only a single proven value; other alternatives stay unknown.
122     *
123     * @param bool $finite Whether all alternatives promise a finite expiration.
124     */
125    public static function fromRanges(TtlRangeSet $ranges, ?string $condition = null, bool $finite = false): self
126    {
127        $lower = $ranges->lowerBound();
128        $upper = $ranges->upperBound();
129
130        if ($lower !== null && $lower === $upper && $condition === null) {
131            return self::known($lower);
132        }
133
134        return new self(
135            TtlEstimateState::Unknown,
136            upperBound: $upper,
137            reason: $condition,
138            lowerBound: $lower,
139            alternatives: count($ranges->intervals) > 1 ? $ranges : null,
140            finite: $finite,
141        );
142    }

175    }

177    /**
178     * Returns the stricter combination of two estimates.
179     *
180     * Invalid dominates, Unconstrained is the identity, and two Known values
181     * keep the earlier one. Every other combination keeps the provable
182     * bounds: the upper bound is the tightest one any side guarantees, the
183     * lower bound survives only when both sides guarantee one, and a
184     * combination pinned to one value without a runtime condition becomes
185     * Known.
186     */
187    public function meet(self $other): self
188    {
189        if ($this->state === TtlEstimateState::Invalid) {
190            return $this;
191        }
192
193        if ($other->state === TtlEstimateState::Invalid) {
194            return $other;
195        }
196
197        if ($this->state === TtlEstimateState::Unconstrained) {
198            return $other;
199        }
200
201        if ($other->state === TtlEstimateState::Unconstrained) {
202            return $this;
203        }
204
205        if ($this->seconds !== null && $other->seconds !== null) {
206            return $this->seconds <= $other->seconds ? $this : $other;
207        }
208
209        return self::fromRanges(
210            $this->ranges()->meet($other->ranges()),
211            $this->condition($other),
212            $this->hasFiniteExpiration() || $other->hasFiniteExpiration(),
213        );
214    }

216    /**
217     * Returns the tightest upper bound two combined estimates guarantee.
218     */
219    public function bound(self $other): ?int
220    {
221        $first = $this->seconds ?? $this->upperBound;
222        $second = $other->seconds ?? $other->upperBound;
223
224        if ($first === null || $second === null) {
225            return $first ?? $second;
226        }
227
228        return min($first, $second);
229    }

232     * Returns the lower bound two combined estimates still guarantee.
233     *
234     * The combination takes the earlier constraint, so the guarantee only
235     * survives when both sides carry one.
236     */
237    public function floor(self $other): ?int
238    {
239        $first = $this->seconds ?? $this->lowerBound;
240        $second = $other->seconds ?? $other->lowerBound;
241
242        if ($first === null || $second === null) {
243            return null;
244        }
245
246        return min($first, $second);
247    }

250     * Returns the runtime condition that survives combining two estimates.
251     */
252    public function condition(self $other): ?string
253    {
254        if ($this->state === TtlEstimateState::Unknown && $this->reason !== null) {
255            return $this->reason;
256        }
257
258        return $other->state === TtlEstimateState::Unknown ? $other->reason : null;

262     * Reports whether two estimates say exactly the same thing.

264    public function equals(self $other): bool
265    {
266        return $this->state === $other->state
267            && $this->seconds === $other->seconds
268            && $this->upperBound === $other->upperBound
269            && $this->lowerBound === $other->lowerBound
270            && $this->alternatives?->bounds() === $other->alternatives?->bounds()

272            && $this->reason === $other->reason;

275    /**
276     * Returns the estimate rendered for human readable output.
277     *
278     * A range keeps what is proven: "30-60s" for both bounds, "30-?s" when
279     * only the lower bound is known — the ? is undetermined, not unlimited —
280     * and "≤60s" when only the upper bound is known.
281     */
282    public function label(): string
283    {
284        if ($this->alternatives !== null) {
285            return $this->alternatives->label();
286        }
287
288        if ($this->seconds !== null) {
289            return $this->seconds.'s';
290        }
291
292        if ($this->lowerBound !== null) {
293            return $this->upperBound === null
294                ? $this->lowerBound.'-?s'
295                : $this->lowerBound.'-'.$this->upperBound.'s';
296        }
297
298        if ($this->upperBound !== null) {
299            return '≤'.$this->upperBound.'s';
300        }
301
302        return $this->state->value;
303    }
