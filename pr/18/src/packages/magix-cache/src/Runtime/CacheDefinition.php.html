26/**
27 * Holds one resolved static cache declaration.
28 *
29 * A definition carries no per-invocation state: arguments and the origin
30 * closure live in CacheInvocation, and the runtime is referenced by name only,
31 * so memoizing a definition never retains a runtime instance. Constraints
32 * from scoped parameters are folded into the policy with the meet, so a
33 * method declaration cannot relax them.
34 *
35 * @internal
36 */
37final readonly class CacheDefinition
38{
39    /**
40     * Effective policy, including the visibility scoped parameters impose.
41     */
42    public CachePolicy $policy;
43
44    /**
45     * Name of the runtime registered for this boundary.
46     */
47    public string $runtime;
48
49    /**
50     * Memoized static strategy recipe; null when absent or invocation-bound.
51     */
52    public ?StrategyDefinition $strategy;

58    /**
59     * Creates a definition from one reflected method and its declarations.
60     *
61     * Static strategy arguments are resolved once here. Parameter bindings
62     * are validated here and evaluated by invocation(), which retains no
63     * caller values on this memoized definition.
64     *
65     * @throws InvalidArgumentException when parameter scopes or configuration bindings conflict
66     * @throws LogicException when the declared strategy cannot be resolved
67     */
68    public function __construct(
69        private ReflectionMethod $method,
70        private string $concreteClass,
71        Cache $declaration,
72        public ?StaleIfError $staleIfError = null,
73        public ?DynamicTtl $dynamicTtl = null,
74        public ?BypassCacheErrors $bypassCacheErrors = null,
75        private ?UseStrategy $useStrategy = null,
76    ) {
77        $visibility = Visibility::Shared;
78
79        foreach ($method->getParameters() as $parameter) {
80            $ignored = $parameter->getAttributes(CacheIgnore::class) !== [];
81            $scopes = $parameter->getAttributes(CacheScope::class);
82
83            if ($scopes === []) {
84                continue;
85            }
86
87            $scope = $scopes[0]->newInstance()->visibility;
88
89            if ($ignored && $scope !== Visibility::NoStore) {
90                throw new InvalidArgumentException('A scoped cache parameter cannot also be ignored unless its scope is NoStore.');
91            }
92
93            $visibility = $visibility->meet($scope);
94        }
95
96        $this->policy = $declaration->policy()->restrictVisibility($visibility);
97        $this->runtime = $declaration->runtime;
98        $this->parameters = new ParameterBindings($method);
99        $bindings = $this->parameters->strategyArguments();
100        (new StrategyBindings())->validate($useStrategy, $bindings);
101        $this->strategy = $bindings === [] ? $useStrategy?->resolve() : null;
102        $this->fingerprint = (new DeclarationFingerprint())->calculate(
103            $method,
104            $this->policy,
105            $staleIfError,
106            $dynamicTtl,
107            $bypassCacheErrors,
108            $useStrategy,
109        );
110    }
111

124    /**
125     * Binds configuration before lookup without retaining invocation values.
126     *
127     * @template T
128     * @param array<array-key, mixed> $arguments
129     * @param Closure(): Cached<T> $origin
130     * @return CacheInvocation<T>
131     * @throws InvalidArgumentException when parameter values do not satisfy their cache constraints
132     * @throws LogicException when the strategy factory does not produce a construction definition
133     */
134    public function invocation(array $arguments, Closure $origin): CacheInvocation
135    {
136        $values = $this->parameters->bind($arguments);
137        $strategy = $this->strategyFor($values);
138
139        return new CacheInvocation(
140            context: $this->context($arguments, $values, $strategy),
141            policy: $values->policy($this->policy),
142            origin: $origin,
143            staleIfError: $this->staleIfError,
144            dynamicTtl: $this->dynamicTtl,
145            bypassCacheErrors: $this->bypassCacheErrors,
146            strategy: $strategy,
147            parameterTtl: $values->ttl,
148        );
149    }
150
151    /**
152     * Builds a fresh recipe only when create() has invocation arguments.
153     *
154     * @throws LogicException when the strategy factory does not produce a construction definition
155     */
156    public function strategyFor(ParameterConfiguration $values): ?StrategyDefinition
157    {
158        return $values->strategyArguments === []
159            ? $this->strategy
160            : $this->useStrategy?->resolve($values->strategyArguments);
161    }
162
163    /**
164     * Includes evaluated settings even when a key reducer omits them.
165     *
166     * @param array<array-key, mixed> $arguments
167     */
168    public function context(array $arguments, ParameterConfiguration $values, ?StrategyDefinition $strategy): CacheKeyContext
169    {
170        $keyArguments = (new CacheKeyArgumentBinder())->bind($this->method, $arguments);
171
172        if ($this->parameters->bindings !== []) {
173            $keyArguments['@configuration'] = [
174                'ttl' => $values->ttl,
175                'tags' => $values->tags,
176                'visibility' => $values->visibility,
177                'strategy' => $strategy,
178            ];
179        }
180
181        return new CacheKeyContext(
182            namespace: '',
183            class: $this->concreteClass,
184            declaringClass: $this->method->getDeclaringClass()->getName(),
185            method: $this->method->getName(),
186            arguments: $keyArguments,
187            version: $this->policy->version,
188            fingerprint: $this->fingerprint,
189        );
190    }
