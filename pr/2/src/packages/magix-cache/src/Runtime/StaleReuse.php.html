12/**
13 * Judges when a retained expired entry may stand in for a failed origin.
14 *
15 * The judgement is pure: it can accept or reject the candidate the lookup
16 * retained, but never replace its expiration or value. Serving a candidate
17 * keeps its expired expiration, so being servable and being storable stay
18 * separate judgements.
19 *
20 * @internal
21 */
22final readonly class StaleReuse
23{
24    /**
25     * Returns the candidate when the failure and the candidate are eligible.
26     *
27     * At judgement time s the candidate must satisfy expiresAt <= s,
28     * s < retainedUntil, and s < expiresAt + maxAge; a candidate exactly at
29     * its retention or age limit is rejected.
30     *
31     * @template T
32     * @param CacheEntry<T>|null $stale
33     * @return CacheEntry<T>|null
34     */
35    public function candidate(?StaleIfError $behavior, ?CacheEntry $stale, Throwable $error, float $now): ?CacheEntry
36    {
37        if ($behavior === null || $stale === null || !$behavior->captures($error)) {
38            return null;
39        }
40
41        if ($stale->expiresAt <= $now && $now < $stale->retainedUntil && $now < $stale->expiresAt + $behavior->maxAge) {
42            return $stale;
43        }
44
45        return null;
46    }
47
48    /**
49     * Returns the physical retention deadline stale handling asks for.
50     *
51     * Extending retention never changes the expiration itself.
52     */
53    public function retention(?StaleIfError $behavior, CacheMetadata $metadata): ?float
54    {
55        if ($behavior === null || $metadata->expiresAt === null) {
56            return null;
57        }
58
59        return $metadata->expiresAt + $behavior->maxAge;
60    }
