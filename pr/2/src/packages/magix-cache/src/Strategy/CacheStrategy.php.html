7use Magix\Cache\Cached;
8use RuntimeException;

10/**
11 * One composable unit of cache behavior with the contract it publishes.
12 *
13 * A strategy participates in the fixed stage order of the runtime — lookup,
14 * origin execution, store — by wrapping the same three operations of the
15 * strategies behind it. Composing strategies yields another CacheStrategy,
16 * so a composition can be composed again; the order is meaningful, because
17 * it decides pre- and post-processing, the capture range of failures, and
18 * which delegations are short-circuited.
19 *
20 * A strategy publishes the effects it guarantees as contract attributes on
21 * its operations, such as Contract\Ttl on fetch(). The analyzer derives the
22 * composed behavior from those contracts and the construction code alone,
23 * so the implementation is free as long as it honors what it declared.
24 */
25interface CacheStrategy
26{
27    /**
28     * Returns the cached value for this operation, or null for a miss.
29     *
30     * Delegating reaches the storage lookup at the end of the chain, which
31     * retains an expired-but-retained entry on the operation as the stale
32     * candidate for the fetch stage.
33     *
34     * @return Cached<mixed>|null
35     * @throws RuntimeException when a delegated read fails with declared behavior
36     */
37    public function get(CacheOperation $operation, NextCacheStrategy $next): ?Cached;
38
39    /**
40     * Produces the value of the boundary with this strategy's constraints.
41     *
42     * Delegating reaches the origin computation at the end of the chain.
43     * Constraints on the produced metadata may only be added through the
44     * metadata meet on the normal origin path, after the delegate returned
45     * with the origin base time stamped on the operation.
46     *
47     * @return Cached<mixed>
48     * @throws RuntimeException when the origin or a delegate fails with declared behavior
49     */
50    public function fetch(CacheOperation $operation, NextCacheStrategy $next): Cached;
51
52    /**
53     * Stores the produced value, or refuses to by not delegating.
54     *
55     * Delegating reaches the re-judged storage write at the end of the
56     * chain. A strategy may extend the physical retention through the
57     * operation before delegating; extending retention never changes the
58     * expiration itself.
59     *
60     * @param Cached<mixed> $result
61     * @throws RuntimeException when a delegated write fails with declared behavior
62     */
63    public function set(CacheOperation $operation, Cached $result, NextCacheStrategy $next): void;
64}
