56        $cap = is_int($policy->ttl) ? $policy->ttl : ($policy->ttl === Ttl::FromUpstream ? $policy->maxTtl : null);
57
58        if ($cap === null || $effect->ttl->state === TtlEstimateState::Invalid) {
59            return null;
60        }
61
62        $composed = $effect->strategy === null ? $constraint->ttl : $constraint->ttl->meet($effect->strategy->ttl);
63
64        if ((!$constraint->hasDependencies && $composed->state === TtlEstimateState::Unconstrained)
65            || !$this->shortens($composed, $cap)) {
66            return null;
67        }

71        return 'local '.$setting.' '.$cap.'s; composed '.$composed->label();

76     *
77     * This numeric comparison is independent of the policy declaration and

80    public function shortens(TtlEstimate $composed, int $cap): bool
81    {
82        if ($composed->state === TtlEstimateState::Invalid

85        }
86
87        $ceiling = $composed->seconds ?? $composed->upperBound ?? $composed->lowerBound;
88
89        return $composed->state === TtlEstimateState::Unconstrained || ($ceiling !== null && $ceiling > $cap);
