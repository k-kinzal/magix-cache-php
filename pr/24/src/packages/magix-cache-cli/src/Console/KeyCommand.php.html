44    /**
45     * Resolves and prints the key of one call.
46     *
47     * A boundary the catalog names but this process cannot key is bad input and
48     * is reported as such. Anything else reaches the console application, which
49     * is already the boundary that renders a failure and sets the exit code.
50     *
51     * @param array<array-key, mixed> $arguments
52     * @param array<array-key, mixed> $path
53     * @param string $namespace Key namespace configured on the runtime that stores this boundary.
54     */
55    public function __invoke(
56        SymfonyStyle $io,
57        #[Argument(description: 'Boundary to key, for example ProductQuery::execute')]
58        string $boundary,
59        #[Argument(description: 'Call arguments in parameter order, each a JSON value')]
60        array $arguments = [],
61        #[Option(description: 'Directory or file to scan, repeatable', name: 'path')]
62        array $path = [],
63        #[Option(description: 'Key namespace configured on CacheRuntime')]
64        string $namespace = CacheKeyResolver::DEFAULT_NAMESPACE,
65    ): int {
66        $matches = $this->catalog->load($path)->search($boundary);
67
68        if (count($matches) !== 1) {
69            $io->error($matches === []
70                ? 'No cache boundary matches "'.$boundary.'".'
71                : 'The reference "'.$boundary.'" matches '.count($matches).' boundaries; use the fully qualified name.');
72
73            return Command::FAILURE;
74        }
75
76        $found = $matches[0];
77
78        if ($found->policy === null) {
79            $io->error($found->id().' declares no #[Cache] on the method or its concrete class, so it has no key.');
80
81            return Command::FAILURE;
82        }
83
84        $values = $this->decode($arguments);
85
86        try {
87            $bound = $this->keys->arguments($found->class, $found->method, $values);
88            $key = $this->keys->resolve($found->class, $found->method, $values, $namespace);
89        } catch (CacheKeyUnresolvable $failure) {
90            $io->error($failure->getMessage());
91
92            return Command::FAILURE;
93        }
94
95        $io->writeln($found->id());
96        $io->writeln('  version    '.$found->policy->version);
96        $io->writeln('  version    '.$found->policy->versionLabel());
97        $io->writeln('  namespace  '.$namespace);
98        $io->writeln('  arguments  '.$this->describeArguments($bound));
99        $io->writeln('  key        '.$key);
100
101        return Command::SUCCESS;
102    }

104    /**
105     * Describes the key arguments after ignored and reduced parameters are applied.
106     *
107     * @param array<string, mixed> $arguments
108     */
109    public function describeArguments(array $arguments): string
110    {
111        return $arguments === [] ? 'none' : implode(', ', array_map(
112            static fn (string $name, mixed $value): string => $name.'='.json_encode($value, JSON_UNESCAPED_SLASHES | JSON_PARTIAL_OUTPUT_ON_ERROR),
113            array_keys($arguments),
114            $arguments,
115        ));
116    }

118    /**
119     * Returns command line arguments decoded as JSON values.
120     *
121     * @param array<array-key, mixed> $arguments
122     * @return list<mixed>
123     */
124    public function decode(array $arguments): array
125    {
126        return array_map(
127            static function (string $argument): mixed {
128                try {
129                    return json_decode($argument, true, 512, JSON_THROW_ON_ERROR);
130                } catch (JsonException) {
131                    return $argument;
132                }
133            },
134            array_values(array_filter($arguments, is_string(...))),
135        );
136    }
