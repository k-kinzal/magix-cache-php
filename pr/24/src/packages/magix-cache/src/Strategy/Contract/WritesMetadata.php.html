9/**
10 * Declares which non-expiration metadata fields fetch() replaces.
11 *
12 * OriginResult::withMetadata() may replace visibility, tags and cacheability,
13 * which no lifetime contract describes. This attribute states which of them an
14 * operation writes, so a strategy that only adjusts expiration keeps the
15 * composed metadata of its dependencies precise instead of erasing it.
16 *
17 * Omitting the attribute declares that the operation preserves every field,
18 * the same promise an omitted Ttl makes about expiration. Supplying it with no
19 * arguments declares that the operation replaces all of them with values the
20 * declaration cannot describe. Naming fields declares exactly those.
21 *
22 * The declaration covers the normal origin path only. As with every contract
23 * here, the implementation is free as long as it honors what it declared: an
24 * operation that writes a field it did not name breaks its own contract.
25 */
26#[Attribute(Attribute::TARGET_METHOD)]
27final readonly class WritesMetadata
28{
29    /**
30     * Whether the operation replaces the visibility of the result.
31     */
32    public bool $visibility;
33
34    /**
35     * Whether the operation replaces the tags of the result.
36     */
37    public bool $tags;
38
39    /**
40     * Whether the operation replaces the cacheability of the result.
41     */
42    public bool $cacheable;
43
44    /**
45     * Declares the metadata fields one operation replaces.
46     *
47     * @param bool|null $visibility Null names no field; every field is null only when no argument was supplied.
48     * @param bool|null $tags Null names no field.
49     * @param bool|null $cacheable Null names no field.
50     */
51    public function __construct(?bool $visibility = null, ?bool $tags = null, ?bool $cacheable = null)
52    {
53        $undescribed = $visibility === null && $tags === null && $cacheable === null;
54
55        $this->visibility = $visibility ?? $undescribed;
56        $this->tags = $tags ?? $undescribed;
57        $this->cacheable = $cacheable ?? $undescribed;
58    }
