43    /**
44     * @param array<string, MetadataFlow> $variables
45     */
46    public function read(Expr $expression, array $variables): MetadataFlow
47    {
48        return match (true) {
49            $expression instanceof Expr\Variable && is_string($expression->name) => $variables[$expression->name] ?? new MetadataFlow('unknown'),
50            $expression instanceof Expr\Assign => $this->read($expression->expr, $variables),
51            $expression instanceof Expr\Ternary => new MetadataFlow('choice', [
52                $this->read($expression->if ?? $expression->cond, $variables), $this->read($expression->else, $variables),
53            ]),
54            $expression instanceof Expr\Match_ => new MetadataFlow('choice', array_map(
55                fn (\PhpParser\Node\MatchArm $arm): MetadataFlow => $this->read($arm->body, $variables),
56                array_values($expression->arms),
57            )),
58            $expression instanceof Expr\BinaryOp\Coalesce => new MetadataFlow('choice', [
59                $this->read($expression->left, $variables), $this->read($expression->right, $variables),
60            ]),
61            $expression instanceof Expr\MethodCall => $this->method($expression, $variables),
62            $expression instanceof Expr\StaticCall => $this->staticCall($expression, $variables),
63            $expression instanceof Expr\ArrayDimFetch => $this->projection($expression, $variables),
64            $expression instanceof Expr\Array_ => new MetadataFlow('collection', array_map(
65                fn (\PhpParser\Node\ArrayItem $item): MetadataFlow => $this->read($item->value, $variables),
66                array_values(array_filter($expression->items, static fn (?\PhpParser\Node\ArrayItem $item): bool => $item !== null && !$item->unpack)),
67            )),
68            $expression instanceof Scalar, $expression instanceof Expr\ConstFetch,
69            $expression instanceof Expr\BinaryOp,
70            $expression instanceof Expr\Cast, $expression instanceof Expr\New_ => new MetadataFlow('none'),
71            $expression instanceof Expr\Throw_ => new MetadataFlow('choice'),
72            default => new MetadataFlow('unknown'),
73        };
74    }

76    /**
77     * @param array<string, MetadataFlow> $variables
78     */
79    public function method(Expr\MethodCall $call, array $variables): MetadataFlow
80    {
81        if (!$call->name instanceof Identifier || $call->isFirstClassCallable()) {
82            return new MetadataFlow('unknown');
83        }
84
85        $name = $call->name->toString();
86
87        if ($name === 'cached' && $call->var instanceof Expr\Variable && $call->var->name === 'this') {
88            return $this->callback($this->argument($call, 0), $variables);
89        }
90
91        $target = (new DependencyReader())->target($call, $this->class, $this->propertyTypes, $this->types($call));
92
93        if ($target !== null && $target[0] !== Cached::class) {
94            return new MetadataFlow('call', target: implode('::', $target));
95        }
96
97        $receiver = $this->read($call->var, $variables);
98
99        if ($name === 'value') {
100            return $target !== null ? new MetadataFlow('none') : $this->extraction($call->var, $variables);
101        }
102
103        return match (true) {
104            $name === 'map', $name === 'unzip' => new MetadataFlow('preserve', [$receiver]),
105            $name === 'flatMap' => new MetadataFlow('meet', [new MetadataFlow('preserve', [$receiver]), $this->callback($this->argument($call, 0), $variables)]),
106            $name === 'flatten' => new MetadataFlow('meet', [new MetadataFlow('preserve', [$receiver]), $this->nested($call->var, $variables)]),
107            $name === 'zip', in_array($name, ['combine2', 'combine3', 'combine4', 'combine5'], true) => new MetadataFlow('meet', [new MetadataFlow('preserve', [$receiver]), ...$this->arguments($call, $variables)]),
108            default => new MetadataFlow('unknown'),
109        };
110    }

138     * @param array<string, MetadataFlow> $variables
139     */
140    public function staticCall(Expr\StaticCall $call, array $variables): MetadataFlow
141    {
142        if (!$call->name instanceof Identifier || !$call->class instanceof Name || $call->isFirstClassCallable()) {
143            return new MetadataFlow('unknown');
144        }
145
146        if ($call->class->toString() !== Cached::class) {
147            $target = (new DependencyReader())->target($call, $this->class, $this->propertyTypes, $this->types($call));
148
149            return $target === null ? new MetadataFlow('unknown') : new MetadataFlow('call', target: implode('::', $target));
150        }
151
152        return match ($call->name->toString()) {
153            'of' => new MetadataFlow('wrap', [$this->metadata($this->argument($call, 1), $variables)], payload: $this->contents($this->argument($call, 0), $variables)),
154            'sequence' => new MetadataFlow('wrap', [$this->collection($this->argument($call, 0), $variables)]),
155            'traverse' => new MetadataFlow('wrap', [$this->traverse($call, $variables)]),
156            default => new MetadataFlow('unknown'),
157        };
158    }

160    /**
161     * Reads callbacks only where the API invokes them, never as independent return paths.
162     *
163     * @param array<string, MetadataFlow> $variables
164     */
165    public function callback(?Expr $expression, array $variables): MetadataFlow
166    {
167        if (!$expression instanceof Expr\ArrowFunction && !$expression instanceof Expr\Closure) {
168            return new MetadataFlow('unknown');
169        }
170
171        if ($expression instanceof Expr\Closure) {
172            $captured = [];
173
174            foreach ($expression->uses as $use) {
175                if (is_string($use->var->name) && isset($variables[$use->var->name])) {
176                    $captured[$use->var->name] = $variables[$use->var->name];
177                }
178            }
179
180            $variables = $captured;
181        }
182
183        foreach ($expression->params as $parameter) {
184            if ($parameter->var instanceof Expr\Variable && is_string($parameter->var->name)) {
185                unset($variables[$parameter->var->name]);
186            }
187        }
188
189        return $expression instanceof Expr\ArrowFunction
190            ? $this->read($expression->expr, $variables)
191            : (new StatementFlowReader($this))->read(array_values($expression->stmts), $variables);
192    }

194    /**
195     * Explicitly reusing a child's metadata preserves that child's alternatives.
196     *
197     * @param array<string, MetadataFlow> $variables
198     */
199    public function metadata(?Expr $expression, array $variables): MetadataFlow
200    {
201        if ($expression === null || ($expression instanceof Expr\ConstFetch && $expression->name->toLowerString() === 'null')) {
202            return new MetadataFlow('none');
203        }
204
205        if ($expression instanceof Expr\PropertyFetch && $expression->name instanceof Identifier && $expression->name->toString() === 'metadata') {
206            return $this->read($expression->var, $variables);
207        }
208
209        return new MetadataFlow('unknown');
210    }

212    /**
213     * Empty traversals contribute no metadata; runtime collections remain unknown.
214     *
215     * @param array<string, MetadataFlow> $variables
216     */
217    public function traverse(Expr\StaticCall $call, array $variables): MetadataFlow
218    {
219        $items = $this->argument($call, 0);
220
221        if (!$items instanceof Expr\Array_) {
222            return new MetadataFlow('unknown');
223        }
224
225        $inputs = [];
226
227        foreach ($items->items as $item) {
228            if ($item->unpack) {
229                return new MetadataFlow('unknown');
230            }
231
232            $inputs[] = $this->callback($this->argument($call, 1), $variables);
233        }
234
235        return new MetadataFlow('meet', $inputs);
236    }

238    /**
239     * One flatten exposes the Cached value created by of() or a map callback.
240     *
241     * @param array<string, MetadataFlow> $variables
242     */
243    public function nested(Expr $expression, array $variables): MetadataFlow
244    {
245        if ($expression instanceof Expr\MethodCall && $expression->name instanceof Identifier && $expression->name->toString() === 'map') {
246            return $this->callback($this->argument($expression, 0), $variables);
247        }
248
249        return $this->read($expression, $variables)->payload ?? new MetadataFlow('unknown');

252    /**
253     * @param array<string, MetadataFlow> $variables
254     */
255    public function projection(Expr\ArrayDimFetch $expression, array $variables): MetadataFlow
256    {
257        $value = $expression->var;
258
259        if ($value instanceof Expr\MethodCall && $value->name instanceof Identifier && $value->name->toString() === 'unzip') {
260            return new MetadataFlow('preserve', [$this->read($value->var, $variables)]);
261        }
262
263        $flow = $this->read($value, $variables);
264
265        return in_array($flow->kind, ['none', 'value'], true) ? $flow : new MetadataFlow('unknown');
266    }

268    /**
269     * @param array<string, MetadataFlow> $variables
270     */
271    public function collection(?Expr $expression, array $variables): MetadataFlow
272    {
273        if ($expression === null) {
274            return new MetadataFlow('unknown');
275        }
276
277        if (!$expression instanceof Expr\Array_) {
278            $flow = $this->read($expression, $variables);
279
280            return $flow->kind === 'collection' ? new MetadataFlow('meet', $flow->inputs) : new MetadataFlow('unknown');
281        }
282
283        $inputs = [];
284
285        foreach ($expression->items as $item) {
286            $inputs[] = $item->unpack ? $this->collection($item->value, $variables) : $this->read($item->value, $variables);
287        }
288
289        return new MetadataFlow('meet', $inputs);
290    }

292    /**
293     * @param array<string, MetadataFlow> $variables
294     * @return list<MetadataFlow>
295     */
296    public function arguments(Expr\MethodCall $call, array $variables): array
297    {
298        $inputs = [];
299
300        foreach ($call->args as $argument) {
301            if ($argument instanceof Arg) {
302                $inputs[] = $argument->unpack ? new MetadataFlow('unknown') : $this->read($argument->value, $variables);
303            }
304        }
305
306        return $inputs;

309    /**
310     * Resolves positional or named arguments without evaluating application code.
311     */
312    public function argument(Expr\MethodCall|Expr\StaticCall $call, int $position): ?Expr
313    {
314        $names = match ($call->name instanceof Identifier ? $call->name->toString() : '') {
315            'cached' => ['compute'],
316            'of' => ['value', 'metadata'],
317            'sequence' => ['items'],
318            'traverse' => ['items', 'transform'],
319            'map', 'flatMap' => ['transform'],
320            default => [],
321        };
322
323        foreach ($call->args as $index => $argument) {
324            if ($argument instanceof Arg && !$argument->unpack
325                && ($argument->name === null ? $index === $position : $argument->name->toString() === ($names[$position] ?? null))) {
326                return $argument->value;
327            }
328        }
329
330        return null;
331    }
