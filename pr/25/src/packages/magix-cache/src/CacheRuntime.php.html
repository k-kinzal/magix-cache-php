31/**
32 * Executes cache boundaries through a fixed sequence of stages.
33 *
34 * The order never depends on how attributes are written: lookup, fresh-hit
35 * judgement, origin execution, metadata overrides at one base time, and a
36 * re-judged store. Each stage runs through the strategy chain of the boundary
37 * — the declared composition, when one exists, in front of the terminal that
38 * owns the stage bodies — so a strategy wraps the stages without being able
39 * to reorder them. Policy, parameter TTL, dynamic TTL and strategies override
40 * in that order, using the single base time taken right after origin success.
41 * The outer strategy runs last on the return path and wins for fields it replaces.
42 * Automatic expiration is validated after the overrides. Only the origin
43 * call is captured as OriginFailure; strategies decide whether to answer it,
44 * and unhandled failures propagate with their original identity.
45 * Every invocation constructs fresh strategy
46 * instances. Only storage reads and writes sit inside the backend bypass
47 * range.
48 */
49final readonly class CacheRuntime
50{
51    private RegisteredExtensions $extensions;
52
53    /**
54     * Creates a cache runtime using a Magix cache implementation.
55     *
56     * @param string $namespace Key namespace that separates this runtime's entries.
57     * @param list<Runtime\Extension\CacheTtlResolver> $ttlResolvers Resolvers referenced by #[DynamicTtl].
58     * @param list<Runtime\Extension\BackendErrorClassifier> $errorClassifiers Classifiers referenced by #[BypassCacheErrors].
59     */
60    public function __construct(
61        private Cache $cache,
62        private ClockInterface $clock = new SystemClock(),
63        private CacheKeyStrategy $keyStrategy = new HashCacheKeyStrategy(),
64        private string $namespace = CacheKeyContext::DEFAULT_NAMESPACE,
65        array $ttlResolvers = [],
66        array $errorClassifiers = [],
67        private ?CacheObserver $observer = null,
68    ) {
69        $this->extensions = new RegisteredExtensions($ttlResolvers, $errorClassifiers);
70    }
71
72    /**
73     * Resolves one cache boundary through the fixed execution stages.
74     *
75     * @template T
76     * @param CacheInvocation<T> $invocation
77     * @return Cached<T>
78     * @throws RuntimeException when an origin failure remains unanswered or a delegated stage fails
79     * @throws LogicException when a referenced extension is not registered or a derived lifetime cannot be derived
80     */
81    public function execute(CacheInvocation $invocation): Cached
82    {
83        $clock = new UnixClock($this->clock);
84        $key = $this->keyStrategy->generate($invocation->context->withNamespace($this->namespace));
85        $operation = new CacheOperation($key, $clock->now(...));
86        $resolver = $this->extensions->ttlResolver($invocation->dynamicTtl);
87        $chain = (new StrategyChain(
88            new GuardedCache($this->cache, $this->observer),
89            $this->extensions->classifier($invocation->bypassCacheErrors),
90            $this->observer,
91        ))->bind($invocation, $resolver);
92
93        if ($invocation->policy->visibility !== Visibility::NoStore) {
94            /** @var CacheRead<T>|null $read */
95            $read = $chain->get($operation);
96
97            if ($read !== null && $read->isFresh($operation->now())) {
98                $this->observer?->observe(CacheEvent::FreshHit, $key);
99
100                return $read->cached;
101            }
102
103            $this->observer?->observe(CacheEvent::Miss, $key);
104        }
105
106        /** @var OriginResult<T>|OriginFailure|CacheAnswer<T> $fetched */
107        $fetched = $chain->fetch($operation);
108
109        if ($fetched instanceof OriginFailure) {
110            throw $fetched->error;
111        }
112
113        if ($fetched instanceof CacheAnswer) {
114            $event = CacheEvent::named($fetched->event);
115
116            if ($event !== null) {
117                $this->observer?->observe($event, $key);
118            }
119
120            return $fetched->cached;
121        }
122
123        $result = $fetched->cached;
124        (new PolicySemantics())->validate($invocation->policy, $result->metadata);
125        $chain->set($operation, new CacheWrite($result));
126
127        return $result;
128    }
129}
