18/**
19 * Follows each return path with its own local bindings.
20 *
21 * A statement this reader does not model cannot decide which value is
22 * returned unless it returns one: everything else can only rebind variables.
23 * Such a statement therefore drops the bindings it writes and the walk goes
24 * on, instead of abandoning the whole method. Only statements that can return
25 * fork the walk into alternatives.
26 */
27final readonly class StatementFlowReader
28{

30     * Creates a statement reader in the declaring method's type context.
31     */

33        private ExpressionFlowReader $expressions,
34        private VariableEffects $variables = new VariableEffects(),

38    /**
39     * @param list<Stmt> $statements
40     * @param array<string, MetadataFlow> $variables
41     * @param int $budget Remaining forks, bounding the alternatives one body may produce.
42     */
43    public function read(array $statements, array $variables = [], int $budget = 24): MetadataFlow
44    {
45        if ($budget < 1) {
46            return MetadataFlow::unknown('control-flow-limit', ($statements[0] ?? null)?->getStartLine() ?? 0);
47        }
48
49        foreach ($statements as $position => $statement) {
50            if ($statement instanceof Stmt\Return_) {
51                return $statement->expr === null ? new MetadataFlow('none') : $this->expressions->read($statement->expr, $variables);
52            }
53
54            if ($this->variables->aliases($statement)) {
55                return MetadataFlow::unknown('aliased-binding', $statement->getStartLine());
56            }
57
58            if ($statement instanceof Stmt\Expression) {
59                $variables = $this->expression($statement->expr, $variables);
60
61                if ($statement->expr instanceof Expr\Throw_) {
62                    return new MetadataFlow('choice');
63                }
64
65                continue;
66            }
67
68            $rest = array_slice($statements, $position + 1);
69
70            if (!$this->affects($statement, $rest)) {
71                $variables = $this->variables->unbind($statement, $variables);
72
73                continue;
74            }
75
76            return $this->fork($statement, $rest, $variables, $budget);
77        }
78
79        return new MetadataFlow('none');
80    }

82    /**
83     * Reports whether a statement can change the value this method returns.
84     *
85     * A statement that neither returns nor writes anything a later statement
86     * reads cannot decide the result, however unmodelled its syntax is.
87     *
88     * @param list<Stmt> $rest
89     */
90    public function affects(Stmt $statement, array $rest): bool
91    {
92        if ($this->returns($statement)) {
93            return true;
94        }
95
96        $written = $this->variables->written($statement);
97
98        if ($written === []) {
99            return false;
100        }
101
102        foreach ($rest as $following) {
103            foreach ($this->variables->reads($following) as $name) {
104                if (in_array($name, $written, true)) {
105                    return true;
106                }
107            }
108        }
109
110        return false;
111    }

113    /**
114     * Returns the alternatives a returning statement opens, or an unread body.
115     *
116     * @param list<Stmt> $rest
117     * @param array<string, MetadataFlow> $variables
118     */
119    public function fork(Stmt $statement, array $rest, array $variables, int $budget): MetadataFlow
120    {
121        $branches = $this->branches($statement);
122
123        if ($branches === null || $this->conditionWrites($statement)) {
124            return MetadataFlow::unknown('unsupported-control-flow', $statement->getStartLine());
125        }
126
127        return new MetadataFlow('choice', array_map(
128            fn (array $branch): MetadataFlow => $this->read([...$branch, ...$rest], $variables, $budget - 1),
129            $branches,
130        ));
131    }

133    /**
134     * Returns the continuations of a statement that can return, or null.
135     *
136     * A loop body is one alternative next to skipping it, because the number
137     * of iterations is not decided here.
138     *
139     * @return list<list<Stmt>>|null
140     */
141    public function branches(Stmt $statement): ?array
142    {
143        if ($statement instanceof Stmt\If_) {
144            return [
145                array_values($statement->stmts),
146                ...array_map(static fn (Stmt\ElseIf_ $branch): array => array_values($branch->stmts), array_values($statement->elseifs)),
147                array_values($statement->else->stmts ?? []),
148            ];
149        }
150
151        if ($statement instanceof Stmt\Switch_) {
152            return $this->cases($statement);
153        }
154
155        if ($statement instanceof Stmt\TryCatch) {
156            $finally = $statement->finally;
157
158            if ($finally !== null && $this->returns($finally)) {
159                return [array_values($finally->stmts)];
160            }
161
162            return array_values([
163                array_values($statement->stmts),
164                ...array_map(static fn (Stmt\Catch_ $catch): array => array_values($catch->stmts), $statement->catches),
165            ]);
166        }
167
168        if ($statement instanceof Stmt\Foreach_ || $statement instanceof Stmt\While_
169            || $statement instanceof Stmt\Do_ || $statement instanceof Stmt\For_) {
170            return [array_values($statement->stmts), []];
171        }
172
173        return null;
174    }

176    /**
177     * Reports whether a statement can return from the declaring method.
178     *
179     * A return inside a nested function belongs to that function, not here.
180     */
181    public function returns(Node $node): bool
182    {
183        $nested = [];
184        $found = [];
185
186        foreach ((new NodeFinder())->find($node, static fn (Node $item): bool => $item instanceof Stmt\Return_
187            || $item instanceof FunctionLike || $item instanceof Stmt\ClassLike) as $item) {
188            if ($item instanceof Stmt\Return_) {
189                $found[] = $item;
190
191                continue;
192            }
193
194            $nested[] = $item;
195        }
196
197        foreach ($found as $return) {
198            if (!$this->inside($return, $nested)) {
199                return true;
200            }
201        }
202
203        return false;
204    }

222    /**
223     * Returns the bindings that survive one expression statement.
224     *
225     * @param array<string, MetadataFlow> $variables
226     * @return array<string, MetadataFlow>
227     */
228    public function expression(Expr $expression, array $variables): array
229    {
230        if (!$expression instanceof Expr\Assign || $this->variables->writes($expression->expr)) {
231            return $this->variables->unbind($expression, $variables);
232        }
233
234        $target = $expression->var;
235
236        if ($target instanceof Expr\Variable && is_string($target->name)) {
237            $variables[$target->name] = $this->expressions->read($expression->expr, $variables);
238
239            return $variables;
240        }
241
242        $root = $target instanceof Expr\ArrayDimFetch ? $this->variables->root($target) : null;
243
244        if ($root === null) {
245            return $this->variables->unbind($expression, $variables);
246        }
247
248        $variables[$root] = $this->variables->extend($variables[$root] ?? null, $this->expressions->read($expression->expr, $variables));
249
250        return $variables;
251    }

253    /**
254     * Conditions can mutate the bindings every continuation reads.
255     */
256    public function conditionWrites(Stmt $statement): bool
257    {
258        $conditions = match (true) {
259            $statement instanceof Stmt\If_ => [$statement->cond, ...array_map(static fn (Stmt\ElseIf_ $branch): Expr => $branch->cond, $statement->elseifs)],
260            $statement instanceof Stmt\Switch_ => [$statement->cond],
261            default => [],
262        };
263
264        foreach ($conditions as $condition) {
265            if ($this->variables->writes($condition)) {
266                return true;
267            }
268        }
269
270        return false;
271    }

273    /**
274     * Each possible entry runs following cases until a break or return.
275     *
276     * @return list<list<Stmt>>
277     */
278    public function cases(Stmt\Switch_ $statement): array
279    {
280        $branches = [];
281        $default = false;
282
283        foreach ($statement->cases as $position => $case) {
284            $default = $default || $case->cond === null;
285            $branch = [];
286
287            foreach (array_slice($statement->cases, $position) as $following) {
288                foreach ($following->stmts as $step) {
289                    if ($step instanceof Stmt\Break_ && ($step->num === null || ($step->num instanceof Node\Scalar\Int_ && $step->num->value === 1))) {
290                        break 2;
291                    }
292
293                    $branch[] = $step;
294                }
295            }
296
297            $branches[] = $branch;
298        }
299
300        if (!$default) {
301            $branches[] = [];
302        }
303
304        return $branches;
305    }
