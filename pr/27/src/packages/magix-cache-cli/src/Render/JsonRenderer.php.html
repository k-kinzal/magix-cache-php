59            'returnType' => $boundary->returnType,
60            'parameters' => $boundary->parameters,
61            'hasDynamicTtl' => $boundary->hasDynamicTtl,
62            'useStrategy' => $boundary->useStrategy === null ? null : [
63                'strategy' => $boundary->useStrategy->strategy,
64                'line' => $boundary->useStrategy->line,
65                'arguments' => array_map((new ArgumentPresentation())->data(...), $boundary->useStrategy->arguments),

67            'via' => $node->via,
68            'calls' => $node->calls,
69            'file' => $boundary->file,
70            'line' => $boundary->line,
71            'policy' => $policy === null ? null : $this->policy($policy),
72            'key' => !$boundary->isCacheBoundary ? null : array_map(
73                static fn ($parameter): array => [
74                    'name' => $parameter->name,
75                    'type' => $parameter->type,
76                    'ignored' => $parameter->ignored,
77                    'scope' => $parameter->scope === null ? null : strtolower($parameter->scope->name),
78                    'reducer' => $parameter->reducer,
79                    'configuration' => $parameter->configuration,
80                ],
81                $boundary->parameters,
82            ),
83            'strategy' => $effect->strategy === null ? null : $this->strategy($effect->strategy),
84            'effective' => $this->effect($node),
85            'composed' => $this->composed($node),
86            'notes' => $node->notes,
87            ...($node->metadataVariants === null ? [] : ['metadataAlternatives' => array_map((new AlternativePresentation())->data(...), $node->metadataVariants)]),
88            'diagnostics' => array_keys($node->diagnostics),
89            'analysisGaps' => array_map($this->gap(...), $node->gaps),
90            'dependencies' => array_map(fn (CacheNode $child): array => $this->tree($child, false), $node->children),
