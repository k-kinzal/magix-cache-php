12final readonly class TreeFilter

15     * @param list<IgnorePattern> $patterns Matching subtrees are removed before promotion.
16     * @param int $depth Display depth counts original calls, including omitted methods.
16     * @param int $depth Display depth counts printed rows, so a compact mode never reaches less far than an expanded one.
17     */
18    public function __construct(private array $patterns = [], private UncachedMode $uncached = UncachedMode::Between, private int $depth = 8)
19    {
20    }
21
22    /**
23     * Returns a forest: an undeclared selected root follows the same rule as descendants.
23     * Returns the analyzed root, retained by every mode, with its selected descendants.
24     *
25     * @return list<CacheNode>
26     */
27    public function apply(CacheNode $node): array
28    {
29        return (new TreeProjection($this->patterns, $this->uncached))->select($node, false, $this->depth)[0];
29        return $this->truncate((new TreeProjection($this->patterns, $this->uncached))->root($node), $this->depth);
30    }
31
32    /**
33     * Limits how far selected rows are printed, after selection read the whole hierarchy.
34     *
35     * @param list<CacheNode> $nodes
36     * @return list<CacheNode>
37     */
38    public function truncate(array $nodes, int $remaining): array
39    {
40        if ($remaining < 0) {
41            return [];
42        }
43
44        return array_map(fn (CacheNode $node): CacheNode => $node->withChildren($this->truncate($node->children, $remaining - 1)), $nodes);
45    }
