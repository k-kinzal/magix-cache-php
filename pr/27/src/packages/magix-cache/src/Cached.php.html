26/**
27 * Carries an evaluated value together with its cache constraints.
28 *
29 * The composition API preserves the constraints of every dependency it is
30 * given: map and unzip keep this value's metadata; flatMap, flatten, zip,
31 * sequence, traverse and combineN meet the metadata of all inputs. Extracting
32 * a value with value() detaches it from its constraints, so a dependency
33 * obtained inside map must be chained with flatMap or combineN instead.
34 *
35 * @template-covariant T
36 */
37final readonly class Cached

93     * Obtains the next result now and composes both sets of constraints.
94     *
95     * @template U
96     * @param Closure(T): self<U> $transform
97     * @return self<U>
98     */
99    public function flatMap(Closure $transform): self
100    {
101        $next = $transform($this->value);
102
103        return new self($next->value, $this->metadata->meet($next->metadata));
104    }

106    /**
107     * Removes exactly one Cached layer and meets both sets of constraints.
108     *
109     * Cached<Cached<U>> becomes Cached<U>; deeper layers remain wrapped.
110     *
111     * @return self<template-type<T, self, 'T'>>
112     * @throws LogicException when the wrapped value is not a Cached result
113     */
114    public function flatten(): self
115    {
116        $inner = $this->value;
117
118        if (!$inner instanceof self) {
119            throw new LogicException('flatten() requires a Cached value inside this Cached result.');
120        }
121
122        return new self($inner->value, $this->metadata->meet($inner->metadata));
123    }

137    /**
138     * Splits a pair into two cached values, each keeping all its constraints.
139     *
140     * Neither projection can recover the weaker metadata from before zip().
141     *
142     * @return array{self<T[0]>, self<T[1]>}
143     * @throws LogicException when the wrapped value is not a two-element list
144     */
145    public function unzip(): array
146    {
147        $pair = $this->value;
148
149        if (!is_array($pair) || !array_is_list($pair) || count($pair) !== 2) {
150            throw new LogicException('unzip() requires a two-element list inside this Cached result.');
151        }
152
153        return [new self($pair[0], $this->metadata), new self($pair[1], $this->metadata)];
154    }

156    /**
157     * Collects cached values into one array and meets every item's constraints.
158     *
159     * The iterable is consumed now, once, preserving its keys. Repeated keys
160     * keep the last value, but the constraints of every item still contribute.
161     * Empty input produces an empty array with top() metadata.
162     *
163     * @template K of array-key
164     * @template U = never
165     * @param iterable<K, self<U>> $items
166     * @return self<array<K, U>>
167     */
168    public static function sequence(iterable $items): self
169    {
170        $values = [];
171        $metadata = CacheMetadata::top();
172
173        foreach ($items as $key => $item) {
174            $values[$key] = $item->value;
175            $metadata = $metadata->meet($item->metadata);
176        }
177
178        return new self($values, $metadata);
179    }

181    /**
182     * Maps each input to a Cached result and collects their met constraints.
183     *
184     * Evaluation is eager, once per item in iteration order, preserving keys.
185     * Repeated keys keep the last value and all constraints. Empty input
186     * produces an empty array with top() metadata without calling transform.
187     *
188     * @template K of array-key
189     * @template V
190     * @template U
191     * @param iterable<K, V> $items
192     * @param Closure(V): self<U> $transform
193     * @return self<array<K, U>>
194     */
195    public static function traverse(iterable $items, Closure $transform): self
196    {
197        $values = [];
198        $metadata = CacheMetadata::top();
199
200        foreach ($items as $key => $item) {
201            $next = $transform($item);
202            $values[$key] = $next->value;
203            $metadata = $metadata->meet($next->metadata);
204        }
205
206        return new self($values, $metadata);
207    }
