14/**
15 * Reads one attribute kind with method-over-class precedence.
16 *
17 * A method-level attribute replaces the class-level attribute of the same
18 * kind as a whole. Class defaults are read from the concrete class only:
19 * parent-class attributes are never inherited implicitly, while an inherited
20 * method keeps the attributes of the method it reflects to.
21 *
22 * @internal
23 */
24final readonly class CacheAttributeReader
25{
26    /**
27     * Returns the effective declaration of one attribute kind, when present.
28     *
29     * @template A of object
30     * @param class-string<A> $attribute
31     * @return A|null
32     * @throws LogicException when the boundary cannot be reflected or repeats a declaration
33     */
34    public function read(object $service, string $methodName, string $attribute): ?object
35    {
36        try {
37            $method = new ReflectionMethod($service, $methodName);
38        } catch (ReflectionException $missing) {
39            throw new LogicException($service::class.'::'.$methodName.' is not a method that can be reflected.', previous: $missing);
40        }
41
42        $attributes = $method->getAttributes($attribute);
43
44        if ($attributes === []) {
45            $attributes = (new ReflectionClass($service))->getAttributes($attribute);
46        }
47
48        if (count($attributes) > 1) {
49            throw new LogicException($service::class.'::'.$methodName.' repeats #['.$attribute.'] on one declaration target.');
50        }
51
52        return $attributes === [] ? null : $attributes[0]->newInstance();
53    }
54}
