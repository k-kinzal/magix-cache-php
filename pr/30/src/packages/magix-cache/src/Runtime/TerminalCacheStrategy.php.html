24/**
25 * Executes the fixed storage and origin stage bodies and the boundary overrides before returning to strategies.
26 *
27 * Lookups expose retained data, origin calls report success or declared
28 * failure, and writes recheck storage eligibility. Strategies own all
29 * fallback decisions. Only the origin call becomes an OriginFailure.
30 *
31 * @internal
32 */
33final readonly class TerminalCacheStrategy implements CacheStrategy

37    /**
38     * Creates the terminal of one runtime execution.
39     *
40     * @param Closure(): Cached<mixed> $origin
41     * @param int<0, max>|null $parameterTtl Validated invocation lifetime.
42     */

44        private GuardedCache $cache,
45        private ?BackendErrorClassifier $classifier,
46        private Closure $origin,
47        private CachePolicy $policy,
48        private ?CacheObserver $observer = null,
49        private ?CacheTtlResolver $ttlResolver = null,
50        private ?int $parameterTtl = null,
51    ) {
52        $this->converter = new CacheEntryConverter();
53    }
54
55    /**
56     * Exposes physically retained data to the lookup chain.
57     *
58     * @return CacheRead<mixed>|null
59     * @throws RuntimeException when the read fails and is not bypassed
60     */
61    #[Override]
62    public function get(CacheOperation $operation, NextCacheStrategy $next): ?CacheRead
63    {
64        $witness = fn (): mixed => ($this->origin)()->value();
65        $entry = $this->cache->lookup($operation->key(), $this->classifier, $witness, $operation->now());
66
67        return $entry === null ? null : new CacheRead($this->converter->toCached($entry), $entry->retainedUntil);
68    }
69
70    /**
71     * Captures declared origin failure or stamps the successful result's time.
72     *
73     * @return OriginResult<mixed>|OriginFailure
74     */
75    #[Override]
76    public function fetch(CacheOperation $operation, NextCacheStrategy $next): OriginResult|OriginFailure
77    {
78        try {
79            $result = ($this->origin)();
80        } catch (RuntimeException $error) {
81            return new OriginFailure($error);
82        }
83
84        $baseTime = $operation->now();
85
86        $metadata = (new OriginOverrides())->apply($this->policy, $this->ttlResolver, $result, $operation->key(), $baseTime, $this->parameterTtl);
87        $result = Cached::of($result->value(), $metadata);
88
89        return new OriginResult($result, $baseTime);
90    }

98    #[Override]

100    {

107        }

110    }
