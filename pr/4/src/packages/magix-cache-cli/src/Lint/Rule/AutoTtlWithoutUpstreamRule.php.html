17/**
18 * Reports derived TTLs whose finite upstream expiration never exists.
19 *
20 * An upstream that is confirmed unconstrained is an error, because applying
21 * the policy throws at runtime. An upstream the analyzer cannot decide is
22 * only a conditional requirement and is reported as a notice.
23 */
24final readonly class AutoTtlWithoutUpstreamRule implements LintRule
25{
26    /**
27     * Creates the rule.
28     */
29    public function __construct(private EffectCalculator $effects = new EffectCalculator())
30    {
31    }
32
33    /**
34     * Returns a finding when a derived TTL has nothing to derive from.
35     *
36     * @return list<Diagnostic>
37     */
38    #[Override]
39    public function check(CacheNode $node, Catalog $catalog): array
40    {
41        unset($catalog);
42        $boundary = $node->boundary;
43        $declared = $boundary->policy?->ttl;
44
45        if (!$declared instanceof Ttl || $boundary->suppliesMetadata || $boundary->hasDynamicTtl) {
46            return [];
47        }
48
49        $upstream = $this->effects->constrain($node->children)->ttl;
50
51        if ($upstream->state === TtlEstimateState::Unconstrained) {
52            return [new Diagnostic(
53                rule: 'auto-ttl-without-upstream',
54                severity: Severity::Error,
55                boundary: $boundary->id(),
56                file: $boundary->file,
57                line: $boundary->line,
58                message: 'Ttl::'.$declared->name.' has no dependency with a finite expiration, so applying the policy throws a LogicException.',
59                hint: 'Declare a fixed ttl, depend on a cached query, or supply CacheMetadata with an expiration.',
60            )];
61        }
62
63        if ($upstream->state === TtlEstimateState::Unknown) {
64            return [new Diagnostic(
65                rule: 'auto-ttl-without-upstream',
66                severity: Severity::Notice,
67                boundary: $boundary->id(),
68                file: $boundary->file,
69                line: $boundary->line,
70                message: 'Ttl::'.$declared->name.' requires a finite upstream expiration at runtime, and whether one exists cannot be determined statically.',
71                hint: 'Confirm the dependencies always carry a finite expiration, or declare a fixed ttl.',
72            )];
73        }
74
75        return [];
76    }
77}
