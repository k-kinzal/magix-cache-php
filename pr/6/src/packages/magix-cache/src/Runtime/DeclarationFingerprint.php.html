25/**
26 * Digests one effective declaration into a canonical cache-key component.
27 *
28 * Any change to the resolved policy, behaviors, scope, or key settings changes
29 * the fingerprint, so entries stored under an older declaration can no longer
30 * answer a lookup.
31 *
32 * @internal
33 */
34final readonly class DeclarationFingerprint
35{
36    /**
37     * Returns a canonical digest of the effective declaration.
38     *
39     * The policy already carries the visibility scoped parameters impose.
40     *
41     * @throws LogicException when the declaration cannot be encoded
42     */
43    public function calculate(
44        ReflectionMethod $method,
45        CachePolicy $policy,
46        ?StaleIfError $staleIfError,
47        ?DynamicTtl $dynamicTtl,
48        ?BypassCacheErrors $bypassCacheErrors,
49        ?UseStrategy $useStrategy = null,
50    ): string {
51        $exceptions = $staleIfError?->exceptions;
52
53        if ($exceptions !== null) {
54            sort($exceptions);
55        }
56
57        $parameters = [];
58
59        foreach ($method->getParameters() as $parameter) {
60            $reducers = $parameter->getAttributes(CacheKey::class);
61
62            $binding = ParameterBinding::read($parameter);
63            $parameters[] = [
64                'name' => $parameter->getName(),
65                'ignored' => $parameter->getAttributes(CacheIgnore::class) !== [],
66                'reducer' => $reducers === [] ? null : $reducers[0]->newInstance()->reduce,
67                ...($binding === null ? [] : ['binding' => $binding]),
68            ];
69        }
70
71        $encoded = json_encode([
72            'ttl' => is_int($policy->ttl) ? $policy->ttl : 'Ttl::'.$policy->ttl->name,
73            'maxTtl' => $policy->maxTtl,
74            'tags' => (new CacheTokenSet())->tags($policy->tags),
75            'visibility' => $policy->visibility->name,
76            'version' => $policy->version,
77            'staleIfError' => $staleIfError === null ? null : [$staleIfError->maxAge, $exceptions],
78            'dynamicTtl' => $dynamicTtl?->resolver,
79            'bypassCacheErrors' => $bypassCacheErrors === null ? null : [$bypassCacheErrors->classifier],
80            'strategy' => $useStrategy === null ? null : [$useStrategy->strategy, $useStrategy->arguments],
81            'parameters' => $parameters,
82        ]);
83
84        if ($encoded === false) {
85            throw new LogicException('The effective cache declaration cannot be encoded for fingerprinting.');
86        }
87
88        return hash('sha256', $encoded);
89    }
90}
