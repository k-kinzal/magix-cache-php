9/**
10 * Carries one inclusive interval of possible lifetimes in the analysis domain.
11 *
12 * Null is a missing proof of a bound, not an unconstrained cache lifetime.
13 */
14final readonly class TtlInterval
15{
16    /**
17     * Records known bounds, including a singleton when they are equal.
18     *
19     * @throws InvalidArgumentException when a bound is negative or the bounds contradict
20     */
21    public function __construct(public ?int $min = null, public ?int $max = null)
22    {
23        if (($min !== null && $min < 0) || ($max !== null && $max < 0)
24            || ($min !== null && $max !== null && $max < $min)) {
25            throw new InvalidArgumentException('A lifetime interval requires non-negative, ordered bounds.');
26        }
27    }
28
29    /**
30     * Returns the image of min(a, b) for every lifetime in the two intervals.
31     */
32    public function meet(self $other): self
33    {
34        return new self(
35            $this->min === null || $other->min === null ? null : min($this->min, $other->min),
36            $this->max === null || $other->max === null ? $this->max ?? $other->max : min($this->max, $other->max),
37        );
38    }
39
40    /**
41     * Orders intervals by their lower bound; a missing bound comes first.
42     */
43    public function compare(self $other): int
44    {
45        if ($this->min === null || $other->min === null) {
46            return ($this->min === null ? 0 : 1) <=> ($other->min === null ? 0 : 1);
47        }
48
49        return $this->min <=> $other->min;
50    }
51
52    /**
53     * Reports whether this interval overlaps a later interval in bound order.
54     */
55    public function overlaps(self $later): bool
56    {
57        return $this->max === null || $later->min === null || $later->min <= $this->max;
58    }
59
60    /**
61     * Returns the enclosing bounds of overlapping alternatives.
62     */
63    public function cover(self $other): self
64    {
65        return new self(
66            $this->min === null || $other->min === null ? null : min($this->min, $other->min),
67            $this->max === null || $other->max === null ? null : max($this->max, $other->max),
68        );
69    }
70
71    /**
72     * Renders a point or interval without a unit, for joining alternatives.
73     */
74    public function label(): string
75    {
76        if ($this->min !== null) {
77            return $this->min === $this->max ? (string) $this->min : $this->min.'-'.($this->max ?? '?');
78        }
79
80        return $this->max === null ? '?' : '≤'.$this->max;
81    }

88    public function bounds(): array
