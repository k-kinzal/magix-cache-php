25    {

32        }

39            $name = $parameter->getName();

41

44            }
45        }
46
47        foreach ($bindings as $target => $source) {
48            $parameter = $parameters[$target] ?? null;
49
50            if ($parameter === null || $parameter->isVariadic() || $parameter->isPassedByReference()) {
51                throw new InvalidArgumentException('StrategyArgument on $'.$source.' requires a declared non-variadic value parameter $'.$target.' on create().');
52            }
53
54            if (isset($occupied[$target])) {
55                throw new InvalidArgumentException('UseStrategy already supplies $'.$target.', also bound from $'.$source.'.');
56            }
57        }
58    }

60    /**
61     * @throws LogicException when the strategy has no public static create()
62     */
63    public function factory(UseStrategy $use): ReflectionMethod
64    {
65        try {
66            $method = new ReflectionMethod($use->strategy, 'create');
67        } catch (ReflectionException $missing) {
68            throw new LogicException($use->strategy.' declares no create().', previous: $missing);
69        }
70
71        if (!$method->isPublic() || !$method->isStatic()) {
72            throw new LogicException($use->strategy.' must declare a public static create().');
73        }
