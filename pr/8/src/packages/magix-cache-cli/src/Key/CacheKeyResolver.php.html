56    /**
57     * Returns the resolved declaration the runtime would derive the key from.
58     *
59     * The declaration is resolved exactly like at runtime, on an instance
60     * created without its constructor, so the key carries the same version
61     * and declaration fingerprint the runtime stores entries under. A
62     * boundary that declares no #[Cache] is the caller's mistake and is
63     * checked before this method runs; here it surfaces as the same
64     * LogicException the runtime raises.
65     *
66     * @throws CacheKeyUnresolvable when the class cannot be loaded in this process
67     */
68    public function definition(string $class, string $method): CacheDefinition
69    {
70        if (!class_exists($class)) {
71            throw new CacheKeyUnresolvable(
72                'The class '.$class.' cannot be loaded. Run magix where the project autoloader can reach it.',
73            );
74        }
75
76        return $this->definitions->resolve((new ReflectionClass($class))->newInstanceWithoutConstructor(), $method);
77    }

79    /**
80     * Returns the arguments after ignored and reduced parameters are applied.
81     *
82     * @param list<mixed> $arguments
83     * @return array<string, mixed>
84     * @throws CacheKeyUnresolvable when the boundary cannot be loaded or the call does not match its parameters
85     */
86    public function arguments(string $class, string $method, array $arguments): array
87    {
88        $reflection = $this->reflect($class, $method);
89        $given = count($arguments);
90        $required = $reflection->getNumberOfRequiredParameters();
91
92        if ($given < $required) {
93            throw new CacheKeyUnresolvable(
94                $reflection->getName().' needs '.$required.' argument(s) to be keyed, but '.$given.' were given.',
95            );
96        }
97
98        if (!$reflection->isVariadic() && $given > $reflection->getNumberOfParameters()) {
99            throw new CacheKeyUnresolvable(
100                $reflection->getName().' declares '.$reflection->getNumberOfParameters().' parameter(s), but '.$given.' arguments were given.',
101            );
102        }
103
104        return $this->binder->bind($reflection, $arguments);
105    }
