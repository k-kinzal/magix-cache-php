46     * @throws JsonException when the tree cannot be encoded as JSON

48    public function __invoke(
49        SymfonyStyle $io,
50        #[Argument(description: 'Boundary or entry point to analyze, for example FooBarQuery::execute or HomeController::index')]
51        string $boundary,
52        #[Option(description: 'Directory or file to scan, repeatable', name: 'path')]

54        #[Option(description: 'Output format: tree, json or mermaid')]
55        string $format = 'tree',
56        #[Option(description: 'Maximum dependency depth to expand')]
57        int $depth = 8,
58    ): int {
59        $catalog = $this->catalog->load($path);
60        $matches = $catalog->search($boundary, includeEntryPoints: true);
61
62        if ($matches === []) {
63            $io->error('No cache boundary or entry point matches "'.$boundary.'".');
64            $io->writeln($this->suggestions($catalog->boundaries()));
65
66            return Command::FAILURE;
67        }
68
69        $tree = new CacheTree($catalog);
70        $nodes = array_map(static fn (BoundaryDeclaration $found): CacheNode => $tree->build($found, $depth), $matches);
71
72        if ($format === 'json') {
73            $io->writeln((new JsonRenderer())->render($nodes));
74
75            return Command::SUCCESS;
76        }
77
78        $renderer = $format === 'mermaid' ? new MermaidRenderer() : new TreeRenderer();
79
80        foreach ($nodes as $node) {
81            $io->writeln($renderer->render($node));
82        }
83
84        return Command::SUCCESS;
85    }

87    /**
88     * Returns the boundaries that can be analyzed instead.

92    public function suggestions(array $boundaries): string
93    {
94        if ($boundaries === []) {
95            return 'No cache boundaries were found. Use --path to point at the directory that contains them.';
96        }
97
98        $known = array_map(static fn (BoundaryDeclaration $found): string => '  '.$found->id(), array_slice($boundaries, 0, 10));
99        $more = count($boundaries) > 10 ? ['  ... and '.(count($boundaries) - 10).' more'] : [];
100
101        return implode("\n", ['Known boundaries:', ...$known, ...$more]);
102    }
