39    /**
40     * Reports every finding and returns a failure when the run is not clean.
41     *
42     * @param array<array-key, mixed> $path
43     * @throws JsonException when the findings cannot be encoded as JSON
44     */
45    public function __invoke(
46        SymfonyStyle $io,
47        #[Option(description: 'Directory or file to scan, repeatable', name: 'path')]
48        array $path = [],
49        #[Option(description: 'Output format: text or json')]
50        string $format = 'text',
51        #[Option(description: 'Fail when warnings are reported as well')]
52        bool $strict = false,
53    ): int {
54        $diagnostics = $this->linter->inspect($this->catalog->load($path));
55
56        if ($format === 'json') {
57            $io->writeln($this->encode($diagnostics));
58        } else {
59            foreach ($diagnostics as $diagnostic) {
60                $io->writeln($this->describe($diagnostic));
61            }
62        }
63
64        $errors = count(array_filter($diagnostics, static fn (Diagnostic $found): bool => $found->severity === Severity::Error));
65        $warnings = count(array_filter($diagnostics, static fn (Diagnostic $found): bool => $found->severity === Severity::Warning));
66
67        if ($format !== 'json') {
68            $io->writeln(count($diagnostics).' findings, '.$errors.' errors, '.$warnings.' warnings');
69        }
70
71        return $errors > 0 || ($strict && $warnings > 0) ? Command::FAILURE : Command::SUCCESS;
72    }

74    /**
75     * Returns one finding rendered for the terminal.
76     */
77    public function describe(Diagnostic $diagnostic): string
78    {
79        $color = match ($diagnostic->severity) {
80            Severity::Error => 'red',
81            Severity::Warning => 'yellow',
82            Severity::Notice => 'cyan',
83        };
84
85        return $diagnostic->file.':'.$diagnostic->line
86            .'  <fg='.$color.'>'.$diagnostic->severity->value.'</>'
87            .'  '.$diagnostic->rule
88            ."\n  ".$diagnostic->boundary.': '.$diagnostic->message
89            .($diagnostic->hint === '' ? '' : "\n  hint: ".$diagnostic->hint)
90            ."\n";
91    }

93    /**
94     * Returns every finding encoded as JSON.
95     *
96     * @param list<Diagnostic> $diagnostics
97     * @throws JsonException when a finding cannot be encoded
98     */
99    public function encode(array $diagnostics): string
100    {
101        $encoded = json_encode(array_map(
102            static fn (Diagnostic $diagnostic): array => [
103                'rule' => $diagnostic->rule,
104                'severity' => $diagnostic->severity->value,
105                'boundary' => $diagnostic->boundary,
106                'file' => $diagnostic->file,
107                'line' => $diagnostic->line,
108                'message' => $diagnostic->message,
109                'hint' => $diagnostic->hint,
110            ],
111            $diagnostics,
112        ), JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_THROW_ON_ERROR);
113
114        return $encoded;
115    }
