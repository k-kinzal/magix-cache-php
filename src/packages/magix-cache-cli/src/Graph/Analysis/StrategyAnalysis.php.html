71        $inputs = [new CacheVariant(new CacheEffect(
72            visibility: $visibility,
73            tags: $tags,
74            visibilityUnknown: $visibilityUnknown,
75            tagsUnknown: $tagsUnknown,
76            analysis: $analysis,
77        ), [$boundary->id()], cached: true)];
