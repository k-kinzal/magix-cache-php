14/**
15 * Holds what the analyzer can prove about the lifetime of one boundary.
16 *
17 * The value keeps "no constraint is declared" apart from "the constraint
18 * depends on runtime values": an unknown lifetime is never converted into a
19 * concrete number or into the absence of a constraint. An unknown lifetime
20 * may still carry proven bounds — a lower bound the constraint never goes
21 * under and an upper bound it never exceeds; a missing bound is
22 * undetermined, not unlimited. The analyzer works with durations and
23 * bounds, a different value domain from the runtime's absolute expirations.
24 */
25final readonly class TtlEstimate implements JsonSerializable
26{
27    /**
28     * Creates an estimate; the shape of each state is fixed.
29     *
30     * @param int|null $seconds Determined lifetime, carried only by Known.
31     * @param int|null $upperBound Bound a stored entry cannot outlive, carried only by Unknown.
32     * @param string|null $reason Why the value holds: a derivation for Known, a runtime condition for Unknown, the problem for Invalid.
33     * @param int|null $lowerBound Bound the lifetime never goes under, carried only by Unknown.
34     * @throws InvalidArgumentException when a field is combined with a state that cannot carry it
35     */
36    public function __construct(
37        public TtlEstimateState $state,
38        public ?int $seconds = null,
39        public ?int $upperBound = null,
40        public ?string $reason = null,
41        public ?int $lowerBound = null,
42    ) {
43        if (($state === TtlEstimateState::Known) !== ($seconds !== null)) {
44            throw new InvalidArgumentException('A lifetime in seconds is carried by a Known estimate and nothing else.');
45        }
46
47        if (($upperBound !== null || $lowerBound !== null) && $state !== TtlEstimateState::Unknown) {
48            throw new InvalidArgumentException('A bound is carried by an Unknown estimate and nothing else.');
49        }
50
51        if ($state === TtlEstimateState::Invalid && $reason === null) {
52            throw new InvalidArgumentException('An Invalid estimate must name its problem.');
53        }
54
55        if (($seconds !== null && $seconds < 0) || ($upperBound !== null && $upperBound < 0) || ($lowerBound !== null && $lowerBound < 0)) {
56            throw new InvalidArgumentException('A lifetime must be zero or greater.');
57        }
58
59        if ($lowerBound !== null && $upperBound !== null && $upperBound < $lowerBound) {
60            throw new InvalidArgumentException('A lifetime upper bound cannot be below the lower bound.');
61        }
62    }

83     * Returns a lifetime that only the runtime can decide.
84     *
85     * @param int|null $upperBound Bound the lifetime cannot exceed if the boundary stores at all.
86     * @param string|null $condition What must hold at runtime for a lifetime to exist.
87     * @param int|null $lowerBound Bound the lifetime never goes under.
88     */
89    public static function unknown(?int $upperBound = null, ?string $condition = null, ?int $lowerBound = null): self
90    {
91        return new self(TtlEstimateState::Unknown, upperBound: $upperBound, reason: $condition, lowerBound: $lowerBound);
92    }

102    /**
103     * Returns the stricter combination of two estimates.
104     *
105     * Invalid dominates, Unconstrained is the identity, and two Known values
106     * keep the earlier one. Every other combination keeps the provable
107     * bounds: the upper bound is the tightest one any side guarantees, the
108     * lower bound survives only when both sides guarantee one, and a
109     * combination pinned to one value without a runtime condition becomes
110     * Known.
111     */
112    public function meet(self $other): self
113    {
114        if ($this->state === TtlEstimateState::Invalid) {
115            return $this;
116        }
117
118        if ($other->state === TtlEstimateState::Invalid) {
119            return $other;
120        }
121
122        if ($this->state === TtlEstimateState::Unconstrained) {
123            return $other;
124        }
125
126        if ($other->state === TtlEstimateState::Unconstrained) {
127            return $this;
128        }
129
130        if ($this->seconds !== null && $other->seconds !== null) {
131            return $this->seconds <= $other->seconds ? $this : $other;
132        }
133
134        $lower = $this->floor($other);
135        $upper = $this->bound($other);
136        $condition = $this->condition($other);
137
138        if ($lower !== null && $lower === $upper && $condition === null) {
139            return self::known($lower);
140        }
141
142        return self::unknown($upper, $condition, $lower);
143    }

145    /**
146     * Returns the tightest upper bound two combined estimates guarantee.
147     */
148    public function bound(self $other): ?int
149    {
150        $first = $this->seconds ?? $this->upperBound;
151        $second = $other->seconds ?? $other->upperBound;
152
153        if ($first === null || $second === null) {
154            return $first ?? $second;
155        }
156
157        return min($first, $second);
158    }
159
160    /**
161     * Returns the lower bound two combined estimates still guarantee.
162     *
163     * The combination takes the earlier constraint, so the guarantee only
164     * survives when both sides carry one.
165     */
166    public function floor(self $other): ?int
167    {
168        $first = $this->seconds ?? $this->lowerBound;
169        $second = $other->seconds ?? $other->lowerBound;
170
171        if ($first === null || $second === null) {
172            return null;
173        }
174
175        return min($first, $second);
176    }

178    /**
179     * Returns the runtime condition that survives combining two estimates.
180     */
181    public function condition(self $other): ?string
182    {
183        if ($this->state === TtlEstimateState::Unknown && $this->reason !== null) {
184            return $this->reason;
185        }
186
187        return $other->state === TtlEstimateState::Unknown ? $other->reason : null;
188    }

190    /**
191     * Reports whether two estimates say exactly the same thing.
192     */
193    public function equals(self $other): bool
194    {
195        return $this->state === $other->state
196            && $this->seconds === $other->seconds
197            && $this->upperBound === $other->upperBound
198            && $this->lowerBound === $other->lowerBound
199            && $this->reason === $other->reason;
200    }

202    /**
203     * Returns the estimate rendered for human readable output.
204     *
205     * A range keeps what is proven: "30-60s" for both bounds, "30-?s" when
206     * only the lower bound is known — the ? is undetermined, not unlimited —
207     * and "≤60s" when only the upper bound is known.
208     */
209    public function label(): string
210    {
211        if ($this->seconds !== null) {
212            return $this->seconds.'s';
213        }
214
215        if ($this->lowerBound !== null) {
216            return $this->upperBound === null
217                ? $this->lowerBound.'-?s'
218                : $this->lowerBound.'-'.$this->upperBound.'s';
219        }
220
221        if ($this->upperBound !== null) {
222            return '≤'.$this->upperBound.'s';
223        }
224
225        return $this->state->value;
226    }
