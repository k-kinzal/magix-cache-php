25    /**
26     * @param list<Stmt> $statements
27     * @param array<string, MetadataFlow> $variables
28     */
29    public function read(array $statements, array $variables = [], int $budget = 8): MetadataFlow
30    {
31        if ($budget < 1) {
32            return new MetadataFlow('unknown');
33        }
34
35        foreach ($statements as $position => $statement) {
36            if ($statement instanceof Stmt\Return_) {
37                return $statement->expr === null ? new MetadataFlow('none') : $this->expressions->read($statement->expr, $variables);
38            }
39
40            if ($statement instanceof Stmt\If_ || $statement instanceof Stmt\Switch_) {
41                if ($this->conditionWrites($statement)) {
42                    return new MetadataFlow('unknown');
43                }
44
45                $rest = array_slice($statements, $position + 1);
46                $branches = $statement instanceof Stmt\If_ ? $this->branches($statement) : $this->cases($statement);
47
48                return new MetadataFlow('choice', array_map(
49                    fn (array $branch): MetadataFlow => $this->read([...$branch, ...$rest], $variables, $budget - 1),
50                    $branches,
51                ));
52            }
53
54            if ($statement instanceof Stmt\Expression) {
55                $expression = $statement->expr;
56
57                if ($this->writes($expression instanceof Expr\Assign ? $expression->expr : $expression)) {
58                    return new MetadataFlow('unknown');
59                }
60
61                if ($expression instanceof Expr\Assign && $expression->var instanceof Expr\Variable && is_string($expression->var->name)) {
62                    $variables[$expression->var->name] = $this->expressions->read($expression->expr, $variables);
63                } elseif ($expression instanceof Expr\Throw_) {
64                    return new MetadataFlow('choice');
65                }
66
67                continue;
68            }
69
70            if (!$statement instanceof Stmt\Nop) {
71                return new MetadataFlow('unknown');
72            }
73        }
74
75        return new MetadataFlow('none');
76    }

78    /**
79     * Conditions can mutate the bindings used by every continuation.
80     */
81    public function conditionWrites(Stmt\If_|Stmt\Switch_ $statement): bool
82    {
83        $conditions = [$statement->cond, ...($statement instanceof Stmt\If_ ? array_map(static fn (Stmt\ElseIf_ $branch): Expr => $branch->cond, $statement->elseifs) : [])];
84
85        foreach ($conditions as $condition) {
86            if ($this->writes($condition)) {
87                return true;
88            }
89        }
90
91        return false;
92    }

95     * Unsupported writes must not reuse a stale local binding.
96     */
97    public function writes(Expr $expression): bool
98    {
99        return (new NodeFinder())->findFirst($expression, static fn (Node $node): bool => $node instanceof Expr\Assign
100            || $node instanceof Expr\AssignRef || $node instanceof Expr\AssignOp
101            || $node instanceof Expr\PreInc || $node instanceof Expr\PostInc
102            || $node instanceof Expr\PreDec || $node instanceof Expr\PostDec) !== null;
103    }

106     * An absent else keeps the continuation reachable with the original bindings.
107     *
108     * @return list<list<Stmt>>
109     */
110    public function branches(Stmt\If_ $statement): array
111    {
112        return [
113            array_values($statement->stmts),
114            ...array_map(static fn (Stmt\ElseIf_ $branch): array => array_values($branch->stmts), array_values($statement->elseifs)),
115            array_values($statement->else->stmts ?? []),
116        ];
117    }

119    /**
120     * Each possible entry runs following cases until a break or return.
121     *
122     * @return list<list<Stmt>>
123     */
124    public function cases(Stmt\Switch_ $statement): array
125    {
126        $branches = [];
127        $default = false;
128
129        foreach ($statement->cases as $position => $case) {
130            $default = $default || $case->cond === null;
131            $branch = [];
132
133            foreach (array_slice($statement->cases, $position) as $following) {
134                foreach ($following->stmts as $step) {
135                    if ($step instanceof Stmt\Break_ && ($step->num === null || ($step->num instanceof Node\Scalar\Int_ && $step->num->value === 1))) {
136                        break 2;
137                    }
138
139                    $branch[] = $step;
140                }
141            }
142
143            $branches[] = $branch;
144        }
145
146        if (!$default) {
147            $branches[] = [];
148        }
149
150        return $branches;
151    }
