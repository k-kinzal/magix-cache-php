30/**
31 * Executes cache boundaries through a fixed sequence of stages.
32 *
33 * The order never depends on how attributes are written: lookup, fresh-hit
34 * judgement, origin execution, constraint application at one base time, and a
35 * re-judged store. Each stage runs through the strategy chain of the boundary
36 * — the declared composition, when one exists, in front of the terminal that
37 * owns the stage bodies — so a strategy wraps the stages without being able
38 * to reorder them. Strategy constraints are met before the policy constraint,
39 * both evaluated at the single base time taken right after the origin
40 * succeeded, so a declared policy can derive its lifetime from what the
41 * strategies guarantee. Only the origin call is captured as OriginFailure;
42 * strategies decide whether to answer it, and unhandled failures propagate
43 * with their original identity. Every invocation constructs fresh strategy
44 * instances. Only storage reads and writes sit inside the backend bypass
45 * range.
46 */
47final readonly class CacheRuntime
48{
49    private RegisteredExtensions $extensions;
50
51    /**
52     * Creates a cache runtime using a Magix cache implementation.
53     *
54     * @param string $namespace Key namespace that separates this runtime's entries.
55     * @param list<Runtime\Extension\CacheTtlResolver> $ttlResolvers Resolvers referenced by #[DynamicTtl].
56     * @param list<Runtime\Extension\BackendErrorClassifier> $errorClassifiers Classifiers referenced by #[BypassCacheErrors].
57     */
58    public function __construct(
59        private Cache $cache,
60        private ClockInterface $clock = new SystemClock(),
61        private CacheKeyStrategy $keyStrategy = new HashCacheKeyStrategy(),
62        private string $namespace = 'magix',
63        array $ttlResolvers = [],
64        array $errorClassifiers = [],
65        private ?CacheObserver $observer = null,
66    ) {
67        $this->extensions = new RegisteredExtensions($ttlResolvers, $errorClassifiers);
68    }
69
70    /**
71     * Resolves one cache boundary through the fixed execution stages.
72     *
73     * @template T
74     * @param CacheInvocation<T> $invocation
75     * @return Cached<T>
76     * @throws RuntimeException when an origin failure remains unanswered or a delegated stage fails
77     * @throws LogicException when a referenced extension is not registered or a derived lifetime cannot be derived
78     */
79    public function execute(CacheInvocation $invocation): Cached
80    {
81        $clock = new UnixClock($this->clock);
82        $key = $this->keyStrategy->generate($invocation->context->withNamespace($this->namespace));
83        $operation = new CacheOperation($key, $clock->now(...));
84        $chain = (new StrategyChain(
85            new GuardedCache($this->cache, $this->observer),
86            $this->extensions->classifier($invocation->bypassCacheErrors),
87            $this->observer,
88        ))->bind($invocation);
89        $resolver = $this->extensions->ttlResolver($invocation->dynamicTtl);
90
91        if ($invocation->policy->visibility !== Visibility::NoStore) {
92            /** @var CacheRead<T>|null $read */
93            $read = $chain->get($operation);
94
95            if ($read !== null && $read->isFresh($operation->now())) {
96                $this->observer?->observe(CacheEvent::FreshHit, $key);
97
98                return $read->cached;
99            }
100
101            $this->observer?->observe(CacheEvent::Miss, $key);
102        }
103
104        /** @var OriginResult<T>|OriginFailure|CacheAnswer<T> $fetched */
105        $fetched = $chain->fetch($operation);
106
107        if ($fetched instanceof OriginFailure) {
108            throw $fetched->error;
109        }
110
111        if ($fetched instanceof CacheAnswer) {
112            $event = CacheEvent::named($fetched->event);
113
114            if ($event !== null) {
115                $this->observer?->observe($event, $key);
116            }
117
118            return $fetched->cached;
119        }
120
121        $metadata = (new OriginConstraints())->apply($invocation->policy, $resolver, $fetched->cached, $key, $fetched->baseTime, $invocation->parameterTtl);
122        $result = Cached::of($fetched->cached->value(), $metadata);
123        $chain->set($operation, new CacheWrite($result));
124
125        return $result;
126    }
127}
