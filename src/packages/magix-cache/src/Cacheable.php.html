19/**
20 * Adds a declarative cache boundary around one method.
21 *
22 * cached() is the anti-corruption layer between PHP and the internal model:
23 * it captures the call site, resolves the static declaration, and delegates
24 * to the runtime the declaration references. Policy, behaviors, and runtime
25 * come from attributes alone; there is no per-call override path.
26 */
27trait Cacheable
28{
29    private static ?CacheDefinitionResolver $magixCacheDefinitions = null;
30
31    /**
32     * Executes the computation once per cache key and propagates its metadata.
33     *
34     * @template T
35     * @param Closure(): Cached<T> $compute
36     * @return Cached<T>
37     * @throws LogicException when the calling boundary cannot be identified or declares no #[Cache]
38     * @throws RuntimeException when the origin computation fails without an eligible stale fallback
39     */
40    final protected function cached(Closure $compute): Cached
41    {
42        $trace = debug_backtrace(DEBUG_BACKTRACE_PROVIDE_OBJECT, 2);
43        $caller = $trace[1] ?? throw new LogicException('Unable to identify the method that called cached().');
44
45        if (($caller['object'] ?? null) !== $this || str_contains($caller['function'], '{closure')) {
46            throw new LogicException('cached() must be called directly from the boundary method, not through a helper or closure.');
47        }
48
49        $definitions = self::$magixCacheDefinitions ??= new CacheDefinitionResolver();
50        $definition = $definitions->resolve($this, $caller['function']);
51
52        return CacheRuntimeRegistry::resolve($definition->runtime)->execute(new CacheInvocation(
53            context: $definition->keyContext($caller['args'] ?? []),
54            policy: $definition->policy,
55            origin: $compute,
56            staleIfError: $definition->staleIfError,
57            dynamicTtl: $definition->dynamicTtl,
58            bypassCacheErrors: $definition->bypassCacheErrors,
59            strategy: $definition->strategy,
60        ));
61    }
62}
