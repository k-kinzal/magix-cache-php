18/**
19 * Holds one resolved static cache declaration.
20 *
21 * A definition carries no per-invocation state: arguments and the origin
22 * closure live in CacheInvocation, and the runtime is referenced by name only,
23 * so memoizing a definition never retains a runtime instance. Constraints
24 * from scoped parameters are folded into the policy with the meet, so a
25 * method declaration cannot relax them.
26 *
27 * @internal
28 */
29final readonly class CacheDefinition
30{
31    /**
32     * Effective policy, including the visibility scoped parameters impose.
33     */
34    public CachePolicy $policy;
35
36    /**
37     * Name of the runtime registered for this boundary.
38     */
39    public string $runtime;
40
41    private string $fingerprint;
42
43    /**
44     * Creates a definition from one reflected method and its declarations.
45     *
46     * @throws InvalidArgumentException when a parameter is both scoped and ignored
47     */
48    public function __construct(
49        private ReflectionMethod $method,
50        private string $concreteClass,
51        Cache $declaration,
52        public ?StaleIfError $staleIfError = null,
53        public ?DynamicTtl $dynamicTtl = null,
54        public ?BypassCacheErrors $bypassCacheErrors = null,
55    ) {
56        $visibility = Visibility::Shared;
57
58        foreach ($method->getParameters() as $parameter) {
59            $ignored = $parameter->getAttributes(CacheIgnore::class) !== [];
60            $scopes = $parameter->getAttributes(CacheScope::class);
61
62            if ($scopes === []) {
63                continue;
64            }
65
66            $scope = $scopes[0]->newInstance()->visibility;
67
68            if ($ignored && $scope !== Visibility::NoStore) {
69                throw new InvalidArgumentException('A scoped cache parameter cannot also be ignored unless its scope is NoStore.');
70            }
71
72            $visibility = $visibility->meet($scope);
73        }
74
75        $this->policy = $declaration->policy()->restrictVisibility($visibility);
76        $this->runtime = $declaration->runtime;
77        $this->fingerprint = (new DeclarationFingerprint())->calculate(
78            $method,
79            $this->policy,
80            $staleIfError,
81            $dynamicTtl,
82            $bypassCacheErrors,
83        );
84    }
85
86    /**
87     * Builds normalized key-strategy input from arguments captured by Cacheable.
88     *
89     * @param array<array-key, mixed> $arguments
90     */
91    public function keyContext(array $arguments): CacheKeyContext
92    {
93        $keyArguments = (new CacheKeyArgumentBinder())->bind($this->method, $arguments);
94
95        return new CacheKeyContext(
96            namespace: '',
97            class: $this->concreteClass,
98            declaringClass: $this->method->getDeclaringClass()->getName(),
99            method: $this->method->getName(),
100            arguments: $keyArguments,
101            version: $this->policy->version,
102            fingerprint: $this->fingerprint,
103        );
104    }
105}
