14 * Supplies runtime dependencies when constructing an invocation's middleware.
15 *
16 * Configuration stays in StrategyDefinition. Missing constructor parameters
17 * typed ClockInterface or CacheObserver receive this runtime's dependencies.
18 * There is no dependency context in the operation or continuation contracts.
19 *
20 * @internal
21 */
22final readonly class StrategyFactory
23{
24    /**
25     * Supplies the clock and optional observer used by this runtime.
26     */
27    public function __construct(private ClockInterface $clock, private ?CacheObserver $observer)
28    {
29    }
30
31    /**
32     * Constructs one middleware with its explicit configuration and dependencies.
33     *
34     * @param class-string<CacheStrategy> $class
35     * @param array<array-key, mixed> $arguments
36     */
37    public function create(string $class, array $arguments): CacheStrategy
38    {
39        $constructor = (new ReflectionClass($class))->getConstructor();
40
41        foreach ($constructor?->getParameters() ?? [] as $parameter) {
42            $name = $parameter->getName();
43            $type = $parameter->getType();
44
45            if (array_key_exists($name, $arguments) || array_key_exists($parameter->getPosition(), $arguments)
46                || !$type instanceof ReflectionNamedType || $parameter->isVariadic()) {
47                continue;
48            }
49
50            if ($type->getName() === ClockInterface::class) {
51                $arguments[$name] = $this->clock;
52            } elseif ($type->getName() === CacheObserver::class && ($this->observer !== null || $type->allowsNull())) {
53                $arguments[$name] = $this->observer;
54            }
55        }
56
57        return new $class(...$arguments);
58    }
