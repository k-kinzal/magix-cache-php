10/**
11 * Declares the lifetime override a strategy makes on the normal origin path.
12 *
13 * The contract is about the candidate constraint the operation contributes,
14 * not about the format of an algorithm: a declaration with bounds promises
15 * that the strategy replaces expiration with a finite lifetime within the
16 * declared bounds relative to the origin base time. This override may extend
17 * an inherited or policy deadline. A missing bound is undetermined, not
18 * unlimited. Omitting the attribute, or supplying no arguments, declares
19 * that the operation preserves the incoming expiration.
20 * With positional alternatives, the constraint belongs to the union of the
21 * declared points and ranges. Alternatives describe possible outcomes, not simultaneous
22 * constraints, and do not prove which runtime condition selects each one.
23 *
24 * The declaration covers the normal origin path only; a stale answer keeps
25 * the expired expiration it was stored with.
26 */
27#[Attribute(Attribute::TARGET_METHOD)]
28final readonly class Ttl
29{
30    /**
31     * Lower bound of the named range; null for alternatives or no constraint.
32     */
33    public int|ConstructorArg|Arg|null $min;
34
35    /**
36     * Upper bound of the named range; null for alternatives or no constraint.
37     */
38    public int|ConstructorArg|Arg|null $max;
39
40    /**
41     * Whether no lifetime arguments were supplied and expiration is preserved.
42     */
43    public bool $unconstrained;
44
45    /**
46     * Normalized positional alternatives; null for named bounds or no constraint.
47     *
48     * @var non-empty-list<int|ConstructorArg|Arg|TtlRange>|null
49     */
50    public ?array $oneOf;
51
52    /**
53     * Declares the candidate lifetime contract of one operation.
54     *
55     * Ttl(30) declares a fixed constraint; Ttl(30, 60, new TtlRange(600, 900))
56     * declares alternatives. Ttl(min: 600, max: 900) is a single range.
57     * Named bounds cannot be mixed with positional alternatives. With no
58     * arguments the operation adds no TTL constraint, as when omitted.
59     *
60     * @param int|ConstructorArg|Arg|TtlRange|null ...$lifetimes Positional fixed lifetimes, references or ranges; alternatively named min/max bounds in seconds. Null is allowed only for an omitted named bound.
61     * @throws InvalidArgumentException when bounds or alternatives are invalid, an unknown name is supplied, or named bounds and positional alternatives are mixed
62     */
63    public function __construct(
64        int|ConstructorArg|Arg|TtlRange|null ...$lifetimes,
65    ) {
66        if (array_is_list($lifetimes)) {
67            $this->min = null;
68            $this->max = null;
69            $this->unconstrained = $lifetimes === [];
70            $this->oneOf = $lifetimes === [] ? null : TtlRange::alternatives($lifetimes);
71
72            return;
73        }
74
75        foreach (array_keys($lifetimes) as $name) {
76            if ($name !== 'min' && $name !== 'max') {
77                throw new InvalidArgumentException('Use positional lifetime alternatives or named min/max bounds, never both.');
78            }
79        }
80
81        $min = $lifetimes['min'] ?? null;
82        $max = $lifetimes['max'] ?? null;
83
84        if ($min instanceof TtlRange || $max instanceof TtlRange) {
85            throw new InvalidArgumentException('A named lifetime bound must be integer seconds or an argument reference.');
86        }
87
88        $range = new TtlRange($min, $max);
89        $this->min = $range->min;
90        $this->max = $range->max;
91        $this->unconstrained = false;
92        $this->oneOf = null;
93    }
94}
